package powerband.archivos;

import java.io.IOException;
import java.nio.ByteBuffer;
import java.nio.channels.FileChannel;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.util.Objects;
import java.util.function.Function;

import powerband.excepciones.ArchivoException;
import powerband.excepciones.ArchivoIncompletoException;
import powerband.excepciones.ArgumentoInvalidoException;
import powerband.logging.LoggerPowerBand;

//Permite facilitar las operaciones con archivos y hacerlo de forma flexible
public final class ArchivoRegistros<T> implements AutoCloseable {

    private final Path ruta;
    private final int tamanoRegistro;
    private final Function<byte[], T> decodificador;
    private final Function<T, byte[]> serializador;
    private FileChannel canalLectura;
    private FileChannel canalEscritura;
    private boolean cerrado;

    public ArchivoRegistros(Path ruta, int tamanoRegistro,
            Function<byte[], T> decodificador,
            Function<T, byte[]> serializador) {
        this.ruta = Objects.requireNonNull(ruta, "ruta");
        if (tamanoRegistro <= 0) {
            throw ArgumentoInvalidoException.tamanoRegistroInvalido();
        }
        this.tamanoRegistro = tamanoRegistro;
        this.decodificador = Objects.requireNonNull(
                decodificador, "decodificador");
        this.serializador = Objects.requireNonNull(
                serializador, "serializador");
    }

    public long cantidadRegistros() throws IOException {
        try {
            FileChannel canal = obtenerCanalLectura();
            long tamanoArchivo = canal.size();
            validarLongitud(tamanoArchivo);
            return tamanoArchivo / tamanoRegistro;
        } catch (ArchivoException | ArgumentoInvalidoException excepcion) {
            if (excepcion instanceof ArchivoException) {
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        excepcion.getMessage(), ruta.toString());
            }
            throw excepcion;
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(
                    ruta.toString(), "contar registros", excepcion);
        }
    }

    public T leer(long posicionBytes) throws IOException {
        validarPosicion(posicionBytes);
        try {
            FileChannel canal = obtenerCanalLectura();
            long tamanoArchivo = canal.size();
            validarLongitud(tamanoArchivo);
            if (posicionBytes > tamanoArchivo - tamanoRegistro) {
                throw ArchivoException.registroFueraDeRango(
                        ruta.toString(), posicionBytes);
            }

            ByteBuffer buffer = ByteBuffer.allocate(tamanoRegistro);
            leerCompletamente(canal, buffer, posicionBytes);
            return decodificar(buffer.array());
        } catch (ArchivoException | ArgumentoInvalidoException excepcion) {
            throw excepcion;
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(ruta.toString(),
                    "leer registro en offset " + posicionBytes, excepcion);
        }
    }

    public void escribir(long posicionBytes, T registro) throws IOException {
        validarPosicion(posicionBytes);
        Objects.requireNonNull(registro, "registro");
        byte[] datos = Objects.requireNonNull(
                serializador.apply(registro), "registro serializado");
        if (datos.length != tamanoRegistro) {
            throw ArgumentoInvalidoException.tamanoRegistroInvalido();
        }

        try {
            FileChannel canal = obtenerCanalEscritura();
            long tamanoArchivo = canal.size();
            validarLongitud(tamanoArchivo);
            if (posicionBytes > tamanoArchivo) {
                throw ArchivoException.escrituraConHueco(
                        ruta.toString(), posicionBytes, tamanoArchivo);
            }
            if (posicionBytes > Long.MAX_VALUE - tamanoRegistro) {
                throw ArchivoException.escrituraConHueco(
                        ruta.toString(), posicionBytes, tamanoArchivo);
            }

            ByteBuffer buffer = ByteBuffer.wrap(datos);
            escribirCompletamente(canal, buffer, posicionBytes);
        } catch (ArchivoException | ArgumentoInvalidoException excepcion) {
            throw excepcion;
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(ruta.toString(),
                    "escribir registro en offset " + posicionBytes,
                    excepcion);
        }
    }

    @Override
    public void close() throws IOException {
        if (cerrado) {
            return;
        }
        cerrado = true;
        IOException fallo = null;
        if (canalLectura != null) {
            try {
                canalLectura.close();
            } catch (IOException excepcion) {
                fallo = excepcion;
            }
        }
        if (canalEscritura != null) {
            try {
                canalEscritura.close();
            } catch (IOException excepcion) {
                if (fallo == null) {
                    fallo = excepcion;
                } else {
                    fallo.addSuppressed(excepcion);
                }
            }
        }
        if (fallo != null) {
            throw ArchivoException.errorES(
                    ruta.toString(), "cerrar el archivo", fallo);
        }
    }

    private FileChannel obtenerCanalLectura() throws IOException {
        validarAbierto();
        if (canalLectura == null) {
            canalLectura = FileChannel.open(ruta, StandardOpenOption.READ);
        }
        return canalLectura;
    }

    private FileChannel obtenerCanalEscritura() throws IOException {
        validarAbierto();
        if (canalEscritura == null) {
            canalEscritura = FileChannel.open(ruta,
                    StandardOpenOption.READ, StandardOpenOption.WRITE);
        }
        return canalEscritura;
    }

    private void validarAbierto() throws IOException {
        if (cerrado) {
            throw ArchivoException.archivoCerrado(ruta.toString());
        }
    }

    private void validarPosicion(long posicionBytes) {
        if (posicionBytes < 0 || posicionBytes % tamanoRegistro != 0) {
            throw ArgumentoInvalidoException.offsetInvalido(
                    tamanoRegistro, posicionBytes);
        }
    }

    private void validarLongitud(long tamanoArchivo) throws IOException {
        long residuo = tamanoArchivo % tamanoRegistro;
        if (residuo != 0) {
            throw new ArchivoIncompletoException(
                    ruta.toString(), tamanoArchivo, tamanoRegistro);
        }
    }

    private T decodificar(byte[] datos) throws IOException {
        try {
            T registro = decodificador.apply(datos);
            if (registro == null) {
                throw ArgumentoInvalidoException.nombreInvalido();
            }
            return registro;
        } catch (ArgumentoInvalidoException excepcion) {
            throw ArchivoException.registroCorrupto(
                    ruta.toString(), excepcion);
        }
    }

    private void leerCompletamente(FileChannel canal, ByteBuffer buffer,
            long posicionBytes) throws IOException {
        while (buffer.hasRemaining()) {
            int leidos = canal.read(buffer, posicionBytes + buffer.position());
            if (leidos < 0) {
                throw ArchivoException.registroFueraDeRango(
                        ruta.toString(), posicionBytes);
            }
            if (leidos == 0) {
                throw ArchivoException.errorES(
                        ruta.toString(), "leer el registro", null);
            }
        }
    }

    private void escribirCompletamente(FileChannel canal, ByteBuffer buffer,
            long posicionBytes) throws IOException {
        while (buffer.hasRemaining()) {
            int escritos = canal.write(buffer, posicionBytes + buffer.position());
            if (escritos == 0) {
                throw ArchivoException.errorES(
                        ruta.toString(), "escribir el registro", null);
            }
        }
    }

}

package powerband.archivos;

import java.io.IOException;
import java.nio.file.Path;
import java.util.Objects;
import java.util.Optional;

import powerband.util.Constantes;

public final class BuscadorIndice {

    private final Path rutaIndice;

    public BuscadorIndice(Path rutaIndice) {
        this.rutaIndice = Objects.requireNonNull(rutaIndice, "rutaIndice");
    }

    public Optional<EntradaIndice> buscar(String nombre) throws IOException {
        ValidacionesGenerales.validarNombre(nombre);
        try (ArchivoRegistros<EntradaIndice> archivo =
                new ArchivoRegistros<>(rutaIndice,
                    Constantes.TAMANO_ENTRADA_INDICE,
                    EntradaIndice::desdeBytes, EntradaIndice::serializar)) {
            long inferior = 0;
            long superior = archivo.cantidadRegistros() - 1;
            while (inferior <= superior) {
                long medio = inferior + (superior - inferior) / 2;
                long posicionBytes =
                        medio * Constantes.TAMANO_ENTRADA_INDICE;
                EntradaIndice entrada = archivo.leer(posicionBytes);
                int comparacion = ComparadorNombres.INSTANCIA.compare(
                        entrada.nombre(), nombre);
                if (comparacion == 0) {
                    return Optional.of(entrada);
                }
                if (comparacion < 0) {
                    inferior = medio + 1;
                } else {
                    superior = medio - 1;
                }
            }
            return Optional.empty();
        }
    }
}

package powerband.archivos;

import java.nio.ByteBuffer;
import java.util.Arrays;
import java.util.Objects;

import powerband.comandos.TipoAccion;
import powerband.sintetizadores.IdSintetizador;
import powerband.util.Constantes;
import powerband.excepciones.ArgumentoInvalidoException;
import powerband.excepciones.tipos.TipoErrorArgumento;
import powerband.excepciones.ConstantesExcepciones;

public final class ComandoGrabado {

    private final long tiempoRelativoMillis;
    private final IdSintetizador idSintetizador;
    private final TipoAccion tipoAccion;
    private final double[] parametros;

    public ComandoGrabado(long tiempoRelativoMillis,
            IdSintetizador idSintetizador, TipoAccion tipoAccion,
            double[] parametros) {
        if (tiempoRelativoMillis < 0) {
            throw ArgumentoInvalidoException.tiempoRelativoNegativo();
        }
        this.tiempoRelativoMillis = tiempoRelativoMillis;
        this.idSintetizador = Objects.requireNonNull(
                idSintetizador, "idSintetizador");
        this.tipoAccion = Objects.requireNonNull(tipoAccion, "tipoAccion");
        Objects.requireNonNull(parametros, "parametros");
        if (parametros.length > Constantes.MAXIMO_PARAMETROS_COMANDO) {
            rechazarCantidadParametros(parametros.length);
        }
        this.parametros = parametros.clone();
    }

    public long tiempoRelativoMillis() {
        return tiempoRelativoMillis;
    }

    public IdSintetizador idSintetizador() {
        return idSintetizador;
    }

    public TipoAccion tipoAccion() {
        return tipoAccion;
    }

    public double[] parametros() {
        return parametros.clone();
    }

    public byte[] serializar() {
        ByteBuffer buffer = ByteBuffer.allocate(
                Constantes.TAMANO_COMANDO_GRABADO)
                .order(Constantes.ORDEN_BYTES_PROGRAMA);
        buffer.putLong(tiempoRelativoMillis);
        buffer.put((byte) idSintetizador.codigo());
        buffer.put((byte) tipoAccion.codigo());
        buffer.put((byte) parametros.length);
        for (int indice = 0;
                indice < Constantes.MAXIMO_PARAMETROS_COMANDO; indice++) {
            buffer.putDouble(indice < parametros.length
                    ? parametros[indice] : 0.0);
        }
        return buffer.array();
    }

    public static ComandoGrabado desdeBytes(byte[] datos) {
        Objects.requireNonNull(datos, "datos");
        if (datos.length != Constantes.TAMANO_COMANDO_GRABADO) {
            throw ArgumentoInvalidoException.detalle(
                    TipoErrorArgumento.TAMANO_REGISTRO_INVALIDO,
                    String.format(ConstantesExcepciones.MSG_COMANDO_LONGITUD,
                            Constantes.TAMANO_COMANDO_GRABADO));
        }

        ByteBuffer buffer = ByteBuffer.wrap(datos)
                .order(Constantes.ORDEN_BYTES_PROGRAMA);
        long tiempoRelativoMillis = buffer.getLong();
        int codigoSintetizador = Byte.toUnsignedInt(buffer.get());
        int codigoAccion = Byte.toUnsignedInt(buffer.get());
        int cantidadParametros = Byte.toUnsignedInt(buffer.get());
        if (cantidadParametros > Constantes.MAXIMO_PARAMETROS_COMANDO) {
            rechazarCantidadParametros(cantidadParametros);
        }

        double[] parametros = new double[cantidadParametros];
        for (int indice = 0;
                indice < Constantes.MAXIMO_PARAMETROS_COMANDO; indice++) {
            double parametro = buffer.getDouble();
            if (indice < cantidadParametros) {
                parametros[indice] = parametro;
            } else if (parametro != 0.0) {
                throw ArgumentoInvalidoException.parametrosNoUtilizados();
            }
        }

        IdSintetizador idSintetizador =
                IdSintetizador.desdeCodigo(codigoSintetizador);
        TipoAccion tipoAccion = TipoAccion.desdeCodigo(codigoAccion);
        return new ComandoGrabado(tiempoRelativoMillis,
                idSintetizador, tipoAccion, parametros);
    }

    private static void rechazarCantidadParametros(int cantidad) {
        throw ArgumentoInvalidoException.parametrosExceso(
                Constantes.MAXIMO_PARAMETROS_COMANDO, cantidad);
    }

    @Override
    public String toString() {
        return "ComandoGrabado{tiempoRelativoMillis=" + tiempoRelativoMillis
                + ", idSintetizador=" + idSintetizador
                + ", tipoAccion=" + tipoAccion
                + ", parametros=" + Arrays.toString(parametros) + '}';
    }
}

        ValidacionesGenerales.validarNombre(nombre);
        if (posicionInicial < 0 || posicionFinal < posicionInicial) {
            throw ArgumentoInvalidoException.intervaloInvalido(
                    posicionInicial, posicionFinal);

        this.posicionInicial = posicionInicial;

        this.fechaEpochMillis = fechaEpochMillis;

    }

        return posicionInicial;
    }

        return posicionFinal;
    }

    public long fechaEpochMillis() {
        return fechaEpochMillis;
    }

        byte[] nombreBytes = ValidacionesGenerales.codificarNombre(nombre);
        ByteBuffer buffer = ByteBuffer.allocate(
                Constantes.TAMANO_ENCABEZADO_GRABACION)
                .order(Constantes.ORDEN_BYTES_PROGRAMA);
        buffer.put(nombreBytes);
        // una lista de bytes en java se inicializa en 0
        buffer.put(new byte[Constantes.LONGITUD_MAXIMA_NOMBRE - buffer.position()]); 
        buffer.putLong(posicionInicial);
        buffer.putLong(posicionFinal);
        buffer.putLong(fechaEpochMillis);
        return buffer.array();
    }

    public static EncabezadoGrabacion desdeBytes(byte[] datos) {
        Objects.requireNonNull(datos, "datos");
        if (datos.length != Constantes.TAMANO_ENCABEZADO_GRABACION) {

                    String.format(
                            ConstantesExcepciones.MSG_ENCABEZADO_LONGITUD,
                            Constantes.TAMANO_ENCABEZADO_GRABACION));
        }

        ByteBuffer buffer = ByteBuffer.wrap(datos)
                .order(Constantes.ORDEN_BYTES_PROGRAMA);
        byte[] nombreBytes = new byte[Constantes.LONGITUD_MAXIMA_NOMBRE];
        buffer.get(nombreBytes);
        String nombre = ValidacionesGenerales.decodificarNombre(nombreBytes);
        long posicionInicial = buffer.getLong();
        long posicionFinal = buffer.getLong();
        long fechaEpochMillis = buffer.getLong();
        return new EncabezadoGrabacion(nombre, posicionInicial,
                posicionFinal, fechaEpochMillis);
    }

    @Override
    public String toString() {
        return "EncabezadoGrabacion{nombre='" + nombre + '\''

                + ", fechaEpochMillis=" + fechaEpochMillis + '}';
    }

package powerband.archivos;

import java.io.IOException;
import java.nio.file.AtomicMoveNotSupportedException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

import powerband.logging.LoggerPowerBand;
import powerband.util.Constantes;
import powerband.excepciones.ArchivoException;
import powerband.excepciones.CantidadInesperadaException;
import powerband.excepciones.NombreDuplicadoException;
import powerband.excepciones.OrdenamientoException;
import powerband.excepciones.tipos.TipoErrorArchivo;
import powerband.excepciones.tipos.TipoErrorOrdenamiento;

public final class ListaGrabaciones {

    private final Path ruta;

    public ListaGrabaciones(Path ruta) {
        this.ruta = Objects.requireNonNull(ruta, "ruta")
                .toAbsolutePath().normalize();
    }

    public void agregar(EntradaListaGrabacion nuevaEntrada)
            throws IOException {
        Objects.requireNonNull(nuevaEntrada, "nuevaEntrada");
        Path rutaTemporal = null;

        try {
            rutaTemporal = Files.createTempFile(
                    ruta.getParent(), ruta.getFileName() + ".", ".tmp");
            try (ArchivoRegistros<EntradaListaGrabacion> archivo =
                        crearArchivoRegistros(ruta);
                    ArchivoRegistros<EntradaListaGrabacion> temporal =
                        crearArchivoRegistros(rutaTemporal)) {
                long cantidad = archivo.cantidadRegistros();
                long posicionInsercion = encontrarPosicionInsercion(
                        archivo, nuevaEntrada, cantidad);
                copiarConInsercion(archivo, temporal, nuevaEntrada, cantidad,
                        posicionInsercion);
                validarListaOrdenada(temporal, cantidad + 1);
            }
            reemplazarArchivo(rutaTemporal);
            rutaTemporal = null;
        } catch (ArchivoException excepcion) {
            eliminarTemporal(rutaTemporal, excepcion);
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    excepcion.getMessage(), ruta.toString());
            throw excepcion;
        } catch (IOException excepcion) {
            ArchivoException fallo = ArchivoException.errorES(
                    ruta.toString(), "agregar una grabación",
                    excepcion);
            eliminarTemporal(rutaTemporal, fallo);

        } catch (RuntimeException excepcion) {

    private long encontrarPosicionInsercion(
            ArchivoRegistros<EntradaListaGrabacion> archivo,
            EntradaListaGrabacion nuevaEntrada, long cantidad)
            throws IOException {
        long posicionInsercion = cantidad;
        for (long indice = 0; indice < cantidad; indice++) {
            EntradaListaGrabacion existente =
                    archivo.leer(posicionBytes(indice));
            if (ComparadorNombres.INSTANCIA.compare(
                    existente.nombre(), nuevaEntrada.nombre()) == 0) {
                throw new NombreDuplicadoException(
                        existente.nombre(), nuevaEntrada.nombre(),
                        ruta.toString());
            }
            if (posicionInsercion == cantidad
                    && compararEntradas(existente, nuevaEntrada) > 0) {
                posicionInsercion = indice;
            }
        }
        return posicionInsercion;
    }

    private void copiarConInsercion(
            ArchivoRegistros<EntradaListaGrabacion> archivo,
            ArchivoRegistros<EntradaListaGrabacion> destino,
            EntradaListaGrabacion nuevaEntrada, long cantidad,
            long posicionInsercion) throws IOException {
        long posicionDestino = 0;
        boolean insertada = false;
        for (long indice = 0; indice < cantidad; indice++) {
            if (!insertada && indice == posicionInsercion) {
                destino.escribir(posicionDestino, nuevaEntrada);
                posicionDestino += Constantes.TAMANO_ENTRADA_LISTA_GRABACION;
                insertada = true;
            }
            destino.escribir(posicionDestino, archivo.leer(posicionBytes(indice)));
            posicionDestino += Constantes.TAMANO_ENTRADA_LISTA_GRABACION;
        }
        if (!insertada) {
            destino.escribir(posicionDestino, nuevaEntrada);
        }
    }

            ArchivoRegistros<EntradaListaGrabacion> listaTemporal,
            long cantidadEsperada) throws IOException {
        if (listaTemporal.cantidadRegistros() != cantidadEsperada) {
            throw new CantidadInesperadaException(
                    ruta.toString(),
                    cantidadEsperada,
                    listaTemporal.cantidadRegistros());
        }

        EntradaListaGrabacion anterior = null;
        for (long indice = 0; indice < cantidadEsperada; indice++) {
            EntradaListaGrabacion actual =
                    listaTemporal.leer(posicionBytes(indice));
            if (anterior != null && compararEntradas(anterior, actual) >= 0) {
                throw new OrdenamientoException(
                        TipoErrorOrdenamiento.INDICE_DESORDENADO,
                        ruta.toString(), actual.nombre());
            }
            anterior = actual;
        }
    }

        return ComparadorNombres.INSTANCIA.compare(
                entrada1.nombre(), entrada2.nombre());
    }

                //En algunos casos como tener 7 registros, la formula se pasa.
                longitudRun = longitudRun > cantidad / 2
                        ? cantidad : longitudRun * 2;

            validarResultado(origenOrdenado, cantidad);
            return entregarResultado(origenOrdenado, temporales);
        } catch (OrdenamientoException excepcion) {

                long longitudActual = Math.min(

    private void mezclarRuns(Path rutaIzquierda, Path rutaDerecha,
            Path rutaSalida, long longitudRun) throws IOException {
        ArchivoRegistros<EntradaIndice> izquierda =
                crearArchivoRegistros(rutaIzquierda);
        ArchivoRegistros<EntradaIndice> derecha =
                crearArchivoRegistros(rutaDerecha);
        ArchivoRegistros<EntradaIndice> salida =
                crearArchivoRegistros(rutaSalida);
        // Cierra los tres archivos automáticamente al terminar
        try (izquierda; derecha; salida) {
            long cantidadIzquierda = izquierda.cantidadRegistros();
            long cantidadDerecha = derecha.cantidadRegistros();
            // Posición actual de lectura en cada entrada y de escritura en la salida
            long indiceIzquierda = 0;
            long indiceDerecha = 0;
            long indiceSalida = 0;

            // Cada vuelta procesa un par de runs (uno de cada archivo)
            while (indiceIzquierda < cantidadIzquierda
                    || indiceDerecha < cantidadDerecha) {
                // Fin (exclusivo) de cada run. Math.min evita pasarse del
                // final del archivo cuando el último run es más corto
                long finRunIzquierdo = Math.min(
                        indiceIzquierda + longitudRun, cantidadIzquierda);
                long finRunDerecho = Math.min(
                        indiceDerecha + longitudRun, cantidadDerecha);

                //Inicia el algoritmo de mezcla
                while (indiceIzquierda < finRunIzquierdo
                        || indiceDerecha < finRunDerecho) {
                    //null si no hay más, guarda lo datos
                    EntradaIndice entradaIzquierda =
                            indiceIzquierda < finRunIzquierdo
                                    ? izquierda.leer(
                                            posicionBytes(indiceIzquierda))
                                    : null;
                    EntradaIndice entradaDerecha =
                            indiceDerecha < finRunDerecho
                                    ? derecha.leer(
                                            posicionBytes(indiceDerecha))
                                    : null;

                    //Comparación segun estado; evita crear otro for
                    int comparacion;
                    if (entradaIzquierda == null) {
                        comparacion = 1;
                    } else if (entradaDerecha == null) {
                        comparacion = -1;
                    } else {
                        comparacion = ComparadorNombres.INSTANCIA.compare(
                                entradaIzquierda.nombre(),
                                entradaDerecha.nombre());
                    }

                    // Empate según el comparador: "Ana" y "ana" son duplicados
                    // (aquí ambas entradas existen, no hay nulls)
                    if (comparacion == 0) {
                        throw new NombreDuplicadoException(
                                entradaIzquierda.nombre(),
                                entradaDerecha.nombre(),
                                rutaIndice.toString());
                    }

                    if (comparacion < 0) {
                        // El menor está a la izquierda
                        salida.escribir(posicionBytes(indiceSalida++),
                                entradaIzquierda);
                        indiceIzquierda++;
                    } else {
                        // El menor está a la derecha (o la izquierda se agotó)
                        salida.escribir(posicionBytes(indiceSalida++),
                                entradaDerecha);
                        indiceDerecha++;
                    }
                }
            }
        }

public final class ReemplazadorIndice {

    private final Path rutaIndice;
    private final Path rutaResultado;

    public ReemplazadorIndice(Path rutaIndice, Path rutaResultado) {
        this.rutaIndice = Objects.requireNonNull(
                rutaIndice, "rutaIndice").toAbsolutePath().normalize();
        this.rutaResultado = Objects.requireNonNull(
                rutaResultado, "rutaResultado").toAbsolutePath().normalize();
    }
    //Realiza validaciones y lo muevo, 
    //En caso de error manda una excepcion
    public void reemplazar() throws IOException {
        try (ArchivoRegistros<EntradaIndice> indice =
                crearArchivoRegistros(rutaIndice)) {
            reemplazar(indice.cantidadRegistros());
        }
    }

    public void reemplazar(long cantidadResultadoEsperada)
            throws IOException {
        try {
            validarArchivos();
            validarResultado(cantidadResultadoEsperada);
            moverResultado();
        } catch (ArchivoException excepcion) {
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    excepcion.getMessage(), rutaIndice.toString());
            throw excepcion;
        } catch (IOException excepcion) {
            ArchivoException fallo = ArchivoException.errorES(
                    rutaIndice.toString(), "reemplazar el índice", excepcion);
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    fallo.getMessage(), rutaIndice.toString());
            throw fallo;
        } catch (RuntimeException excepcion) {
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    "No se pudo reemplazar el índice " + rutaIndice + ": "
                            + excepcion.getMessage(),
                    rutaIndice.toString());
            throw excepcion;
        }
    }
    //Validaciones de archivo, directorio etc
    private void validarArchivos() throws IOException {
        if (!Files.isRegularFile(rutaIndice)) {
            throw ArchivoException.rutaNoRegular(rutaIndice.toString());
        }
        if (!Files.isRegularFile(rutaResultado)) {
            throw ArchivoException.rutaNoRegular(rutaResultado.toString());
        }

        Path directorioIndice = rutaIndice.getParent();
        Path directorioResultado = rutaResultado.getParent();
        if (directorioIndice == null
                || !directorioIndice.equals(directorioResultado)) {
            throw ArchivoException.rutaNoRegular(rutaResultado.toString());
        }
        if (rutaIndice.equals(rutaResultado)) {
            throw ArchivoException.rutaNoRegular(rutaResultado.toString());
        }
    }
    //Validacion de duplicados y ordenados.
    private void validarResultado(long cantidadResultadoEsperada)
            throws IOException {
        long cantidadResultado;
        try (ArchivoRegistros<EntradaIndice> indice =
                    crearArchivoRegistros(rutaIndice);
                ArchivoRegistros<EntradaIndice> resultado =
                    crearArchivoRegistros(rutaResultado)) {
            cantidadResultado = resultado.cantidadRegistros();
        }

        if (cantidadResultado != cantidadResultadoEsperada) {
            throw new CantidadInesperadaException(
                    rutaResultado.toString(), cantidadResultadoEsperada,
                    cantidadResultado);
        }

        EntradaIndice anterior = null;
        try (ArchivoRegistros<EntradaIndice> resultado =
                crearArchivoRegistros(rutaResultado)) {
            for (long indice = 0; indice < cantidadResultado; indice++) {
                EntradaIndice actual = resultado.leer(
                        indice * Constantes.TAMANO_ENTRADA_INDICE);
                if (anterior != null
                        && ComparadorNombres.INSTANCIA.compare(
                                anterior.nombre(), actual.nombre()) >= 0) {
                    throw new OrdenamientoException(
                            TipoErrorOrdenamiento.INDICE_DESORDENADO,
                            rutaResultado.toString(), actual.nombre());
                }
                anterior = actual;
            }
        }
    }
    //Mueve del resultado al indice, 
    //Hace un remplazo completo
    //se ejecuta de forma atomica, en escencia eso significa 
    //que el archivo no queda en ningun estado corrompido si sucede algun error
    private void moverResultado() throws IOException {
        try {
            Files.move(rutaResultado, rutaIndice,
                    StandardCopyOption.ATOMIC_MOVE,
                    StandardCopyOption.REPLACE_EXISTING);
        } catch (AtomicMoveNotSupportedException excepcion) {
            throw new ArchivoException(
                    TipoErrorArchivo.REEMPLAZO_NO_ATOMICO,
                    rutaIndice.toString(),
                    excepcion);
        } catch (java.nio.file.AccessDeniedException excepcion) {
            try {
                Files.move(rutaResultado, rutaIndice,
                        StandardCopyOption.REPLACE_EXISTING);
            } catch (IOException fallback) {
                excepcion.addSuppressed(fallback);
                throw new ArchivoException(
                        TipoErrorArchivo.REEMPLAZO_NO_ATOMICO,
                        rutaIndice.toString(),
                        excepcion);
            }
        } catch (IOException excepcion) {
            throw new ArchivoException(
                    TipoErrorArchivo.REEMPLAZO_NO_ATOMICO,
                    rutaIndice.toString(),
                    excepcion);
        }
    }

    static void validarNombre(String nombre) {
        byte[] bytes = codificarNombre(nombre);
        if (bytes.length == 0
                || bytes.length > Constantes.LONGITUD_MAXIMA_NOMBRE) {
            throw ArgumentoInvalidoException.nombreInvalido();
        }
        if (nombre.indexOf('\0') >= 0) {
            throw ArgumentoInvalidoException.nombreConNulos();
        }
    }

    static byte[] codificarNombre(String nombre) {
        if (nombre == null) {
            throw ArgumentoInvalidoException.nombreNulo();
        }
        try {
            ByteBuffer codificado = StandardCharsets.UTF_8.newEncoder()
                    .onMalformedInput(CodingErrorAction.REPORT)
                    .onUnmappableCharacter(CodingErrorAction.REPORT)
                    .encode(CharBuffer.wrap(nombre));
            byte[] bytes = new byte[codificado.remaining()];
            codificado.get(bytes);
            return bytes;
        } catch (CharacterCodingException excepcion) {
            throw ArgumentoInvalidoException.nombreUtf8Invalido(excepcion);
        }
    }

    static String decodificarNombre(byte[] bytes) {
        int longitud = 0;
        while (longitud < bytes.length && bytes[longitud] != 0) {
            longitud++;
        }
        for (int indice = longitud; indice < bytes.length; indice++) {
            if (bytes[indice] != 0) {
                throw ArgumentoInvalidoException.rellenoNombreInvalido();
            }
        }
        if (longitud == 0) {
            throw ArgumentoInvalidoException.nombreVacio();
        }
        try {
            return StandardCharsets.UTF_8.newDecoder()
                    .onMalformedInput(CodingErrorAction.REPORT)
                    .onUnmappableCharacter(CodingErrorAction.REPORT)
                    .decode(ByteBuffer.wrap(bytes, 0, longitud))
                    .toString();
        } catch (CharacterCodingException excepcion) {
            throw ArgumentoInvalidoException.nombreCodificadoInvalido(excepcion);
        }
    }
}

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

import powerband.logging.LoggerPowerBand;

public final class BusComandos {

    private final List<ObservadorComandos> observadores = new ArrayList<>();

    public void registrarObservador(ObservadorComandos observador) {
        Objects.requireNonNull(observador, "observador");
        synchronized (observadores) {
            boolean yaRegistrado = observadores.stream()
                    .anyMatch(registrado -> registrado == observador);
            if (!yaRegistrado) {
                observadores.add(observador);
            }
        }
    }

    public void quitarObservador(ObservadorComandos observador) {
        Objects.requireNonNull(observador, "observador");
        synchronized (observadores) {
            observadores.removeIf(registrado -> registrado == observador);
        }
    }

    public void publicar(Comando comando, long momentoNanos) {
        Objects.requireNonNull(comando, "comando");

        List<ObservadorComandos> copiaObservadores;
        synchronized (observadores) {
            copiaObservadores = List.copyOf(observadores);
        }

        List<RuntimeException> fallos = new ArrayList<>();
        for (ObservadorComandos observador : copiaObservadores) {
            try {
                boolean procesado =
                        observador.comandoEnviado(comando, momentoNanos);
                if (!procesado) {
                    fallos.add(new IllegalStateException(
                            "El observador no pudo procesar el comando"));
                }
            } catch (RuntimeException fallo) {
                fallos.add(fallo);
            }
        }

        LoggerPowerBand logger = LoggerPowerBand.getInstance();
        for (RuntimeException fallo : fallos) {
            logger.registrarError("Fallo al notificar un comando a un observador: "
                    + fallo);
        }
    }
}

                            + idSintetizador);

        return new ComandoReconstruido(destino, grabado.tipoAccion(),
                grabado.parametros());
    }
}

                return false;
            }
            destinos.add(sintetizador);
        }
        instrumento.reemplazarSintetizadores(destinos);
        asignacionesSintetizador.put(
                instrumentoId, List.copyOf(sintetizadorIds));
        return true;
    }

    @Override
    public void iniciarGrabacion() {
        synchronized (this) {
            if (estado != EstadoControlador.CONECTADO
                    || grabando || reproduciendo) {
                throw new IllegalStateException(
                        "No se puede iniciar la grabación en el estado actual");
            }
        }
        try {
            grabador.iniciar();
            busComandos.registrarObservador(grabador);
            synchronized (this) {
                grabando = true;
            }
            cambiarEstado(EstadoControlador.GRABANDO);
        } catch (IOException | RuntimeException excepcion) {
            notificarError("No se pudo iniciar la grabación", excepcion);
            throw excepcion instanceof RuntimeException runtime
                    ? runtime
                    : new IllegalStateException(
                            "No se pudo iniciar la grabación", excepcion);
        }
    }

    @Override
    public void detenerGrabacion(String nombre) {
        synchronized (this) {
            if (!grabando) {
                throw new IllegalStateException(
                        "No hay una grabación activa");
            }
        }
        try {
            finalizador.finalizar(nombre);
            busComandos.quitarObservador(grabador);
            synchronized (this) {
                grabando = false;
            }
            cambiarEstado(EstadoControlador.CONECTADO);
        } catch (IOException | RuntimeException excepcion) {
            synchronized (this) {
                grabando = false;
            }
            busComandos.quitarObservador(grabador);
            notificarError("No se pudo finalizar la grabación", excepcion);
            throw excepcion instanceof RuntimeException runtime
                    ? runtime
                    : new IllegalStateException(
                            "No se pudo finalizar la grabación", excepcion);
        }
    }

    @Override
    public List<String> listarSesiones() {
        try {
            List<String> nombres = new ArrayList<>();
            for (EntradaListaGrabacion entrada : cargador.listar()) {
                nombres.add(entrada.nombre());
            }
            return List.copyOf(nombres);
        } catch (IOException excepcion) {
            notificarError("No se pudieron listar las grabaciones", excepcion);
            throw new IllegalStateException(
                    "No se pudieron listar las grabaciones", excepcion);
        }
    }

    @Override
    public List<String> buscarSesiones(String nombre) {
        ValidacionesGenerales.validarNombre(nombre);
        String filtro = nombre.toLowerCase(Locale.ROOT);
        List<String> coincidencias = new ArrayList<>();
        for (String sesion : listarSesiones()) {
            if (sesion.toLowerCase(Locale.ROOT).contains(filtro)) {
                coincidencias.add(sesion);
            }
        }
        return List.copyOf(coincidencias);
    }

    @Override
    public boolean reproducirSesion(String nombre) {
        ValidacionesGenerales.validarNombre(nombre);
        synchronized (this) {
            if (estado != EstadoControlador.CONECTADO
                    || grabando || reproduciendo) {
                notificarError(
                        "No se puede reproducir en el estado actual",
                        new IllegalStateException(
                                "Estado inválido para reproducir"));
                return false;
            }
        }
        try {
            Optional<GrabacionAbierta> grabacion = cargador.abrir(nombre);
            if (grabacion.isEmpty()) {
                notificarError(
                        "La sesión no existe: " + nombre,
                        new IllegalArgumentException(
                                "La sesión no existe"));
                return false;
            }
            synchronized (this) {
                reproduciendo = true;
            }
            cambiarEstado(EstadoControlador.REPRODUCIENDO);
            Thread hilo = new Thread(
                    () -> reproducirEnSegundoPlano(grabacion.orElseThrow()),
                    "reproduccion-" + nombre);
            hilo.setDaemon(true);
            hiloReproduccion = hilo;
            hilo.start();
            return true;
        } catch (IOException | RuntimeException excepcion) {
            synchronized (this) {
                reproduciendo = false;
            }
            notificarError("No se pudo reproducir la sesión", excepcion);
            return false;
        }
    }

    @Override
    public void registrarObservadorEstado(ObservadorEstado observador) {
        Objects.requireNonNull(observador, "observador");
        synchronized (observadores) {
            boolean registrado = observadores.stream()
                    .anyMatch(actual -> actual == observador);
            if (!registrado) {
                observadores.add(observador);
            }
        }
    }

    @Override
    public void retirarObservadorEstado(ObservadorEstado observador) {
        Objects.requireNonNull(observador, "observador");
        synchronized (observadores) {
            observadores.removeIf(actual -> actual == observador);
        }
    }

    private void reproducirEnSegundoPlano(GrabacionAbierta grabacion) {
        try {
            reproductor.reproducir(grabacion);
        } catch (IOException | InterruptedException | RuntimeException
                | Error excepcion) {
            if (excepcion instanceof InterruptedException) {
                Thread.currentThread().interrupt();
            }
            notificarError(
                    "Error durante la reproducción",
                    excepcion instanceof Throwable throwable
                            ? throwable
                            : new IllegalStateException(excepcion));
        } finally {
            synchronized (this) {
                reproduciendo = false;
                hiloReproduccion = null;
                estado = EstadoControlador.CONECTADO;
            }
            cambiarEstado(EstadoControlador.CONECTADO);
        }
    }

    private void asegurarSintetizadores() throws IOException {
        if (sintetizadores.size() == 3) {
            return;
        }
        BandClient nuevoCliente = new BandClient(
                Configuraciones.getInstance());
        Map<IdSintetizador, Sintetizador> nuevos =
                new LinkedHashMap<>();
        nuevos.put(IdSintetizador.PIANO,
                new PianoSintetizador(nuevoCliente));
        nuevos.put(IdSintetizador.BAJO,
                new BajoSintetizador(nuevoCliente));
        nuevos.put(IdSintetizador.ORGANO,
                new OrganoSintetizador(nuevoCliente));
        synchronized (this) {
            bandClient = nuevoCliente;
            sintetizadores.clear();
            sintetizadores.putAll(nuevos);
        }
        for (Map.Entry<String, List<IdSintetizador>> asignacion
                : asignacionesSintetizador.entrySet()) {
            Instrumento instrumento = instrumentos.get(asignacion.getKey());
            if (instrumento != null) {
                List<Sintetizador> destinos = new ArrayList<>();
                for (IdSintetizador id : asignacion.getValue()) {
                    Sintetizador destino = sintetizadores.get(id);
                    if (destino != null) {
                        destinos.add(destino);
                    }
                }
                instrumento.reemplazarSintetizadores(destinos);
            }
        }
    }

    private void entradaDesconectada(Entrada entrada, String motivo) {
        String nombre = entrada == null ? "desconocida" : entrada.nombre();
        String mensaje = "Desconexión de entrada " + nombre
                + ": " + (motivo == null ? "dispositivo desconectado"
                        : motivo);
        notificarError(mensaje,
                new IllegalStateException(mensaje));
        desconectarRecursos();
        cambiarEstado(EstadoControlador.INICIAL);
    }

    private void desconectarPorFallo(
            String mensaje, Throwable causa) {
        LoggerPowerBand.getInstance().registrarError(
                mensaje + ": " + causa);
        notificarError(mensaje, causa);
        desconectarRecursos();
        cambiarEstado(EstadoControlador.INICIAL);
    }

    private void desconectarRecursos() {
        entradaMicrofono.detener();
        entradaControl.detener();
        entradaTableta.detener();
        if (reproductor.detener()) {
            Thread hilo = hiloReproduccion;
            if (hilo != null) {
                hilo.interrupt();
            }
        }
        for (Sintetizador sintetizador : sintetizadores.values()) {
            try {
                sintetizador.liberarTodasLasNotas();
            } catch (RuntimeException excepcion) {
                notificarError(
                        "No se pudieron liberar las notas sostenidas",
                        excepcion);
            }
        }
        if (grabando) {
            busComandos.quitarObservador(grabador);
            try {
                grabador.detener();
            } catch (IOException | RuntimeException excepcion) {
                notificarError(
                        "No se pudo detener la grabación al desconectar",
                        excepcion);
            }
        }
        if (bandClient != null) {
            try {
                bandClient.close();
            } catch (IOException excepcion) {
                notificarError("No se pudo cerrar BandClient", excepcion);
            } finally {
                bandClient = null;
                sintetizadores.clear();
            }
        }
        synchronized (this) {
            grabando = false;
            reproduciendo = false;
            estado = EstadoControlador.INICIAL;
        }
    }

    private void cambiarEstado(EstadoControlador nuevoEstado) {
        synchronized (this) {
            estado = nuevoEstado;
        }
        List<ObservadorEstado> copia;
        synchronized (observadores) {
            copia = List.copyOf(observadores);
        }
        for (ObservadorEstado observador : copia) {
            try {
                observador.estadoCambiado(nuevoEstado);
            } catch (RuntimeException excepcion) {
                LoggerPowerBand.getInstance().registrarError(
                        "Fallo al notificar un estado a un observador: "
                                + excepcion);
            }
        }
    }

    private void notificarError(String mensaje, Throwable causa) {
        LoggerPowerBand.getInstance().registrarError(
                mensaje + ": " + causa);
        List<ObservadorEstado> copia;
        synchronized (observadores) {
            copia = List.copyOf(observadores);
        }
        for (ObservadorEstado observador : copia) {
            try {
                observador.errorNotificado(mensaje, causa);
            } catch (RuntimeException excepcion) {
                LoggerPowerBand.getInstance().registrarError(
                        "Fallo al notificar un error a un observador: "
                                + excepcion);
            }
        }
    }

        componentes.tableta = new Tableta(
                null, componentes.busComandos);
        componentes.entradaMicrofono =
                new EntradaMicrofono(componentes.microfono);
        componentes.entradaControl =
                new EntradaControl(componentes.control);
        componentes.entradaTableta =
                new EntradaTableta(componentes.tableta);
        componentes.sintetizadores = new LinkedHashMap<>();
        componentes.archivos = ArchivosGrabacion.inicializar();
        componentes.grabador = new Grabador(
                componentes.archivos.directorio());
        componentes.cargador = new CargadorGrabaciones(

                fabrica::reconstruir,

        synchronized (observadores) {
            copiaObservadores = List.copyOf(observadores);
        }

        List<RuntimeException> fallos = new ArrayList<>();
        for (ObservadorEstado observador : copiaObservadores) {
            try {

    private void notificarError(String mensaje, Throwable causa) {
        List<ObservadorEstado> copiaObservadores;
        synchronized (observadores) {
            copiaObservadores = List.copyOf(observadores);

        List<RuntimeException> fallos = new ArrayList<>();
        for (ObservadorEstado observador : copiaObservadores) {

                fallos.add(fallo);

package powerband.grabaciones;

import java.io.IOException;
import java.nio.file.FileAlreadyExistsException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Objects;

import powerband.logging.LoggerPowerBand;
import powerband.util.Constantes;
import powerband.excepciones.ArchivoException;

public final class ArchivosGrabacion {

    private final Path directorio;
    private final Path archivoMusica;
    private final Path archivoLista;
    private final Path archivoIndice;

    private ArchivosGrabacion(Path directorio, Path archivoMusica,
            Path archivoLista, Path archivoIndice) {
        this.directorio = directorio;
        this.archivoMusica = archivoMusica;
        this.archivoLista = archivoLista;
        this.archivoIndice = archivoIndice;
    }

    public static ArchivosGrabacion inicializar() throws IOException {
        return inicializar(Path.of(Constantes.RUTA_GRABACIONES));
    }
    //Se encarga de crear ese archivo si fuera necesario
    public static ArchivosGrabacion inicializar(Path directorio)
            throws IOException {
        Path rutaDirectorio = Objects.requireNonNull(
                directorio, "directorio").toAbsolutePath().normalize();
        Path archivoMusica = rutaDirectorio.resolve(
                Constantes.NOMBRE_ARCHIVO_GRABACIONES_MUSICA);
        Path archivoLista = rutaDirectorio.resolve(
                Constantes.NOMBRE_ARCHIVO_GRABACIONES_LISTA);
        Path archivoIndice = rutaDirectorio.resolve(
                Constantes.NOMBRE_ARCHIVO_GRABACIONES_INDICE);

        try {
            Files.createDirectories(rutaDirectorio);
            crearSiFalta(archivoMusica);
            crearSiFalta(archivoLista);
            crearSiFalta(archivoIndice);
        } catch (IOException excepcion) {
            ArchivoException fallo = ArchivoException.errorES(
                    rutaDirectorio.toString(),
                    "preparar los archivos de grabación", excepcion);
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    fallo.getMessage(), rutaDirectorio.toString());
            throw fallo;
        }

        return new ArchivosGrabacion(rutaDirectorio, archivoMusica,
                archivoLista, archivoIndice);
    }

    public Path directorio() {
        return directorio;
    }

    public Path archivoMusica() {
        return archivoMusica;
    }

    public Path archivoLista() {
        return archivoLista;
    }

    public Path archivoIndice() {
        return archivoIndice;
    }

    private static void crearSiFalta(Path ruta) throws IOException {
        try {
            Files.createFile(ruta);
            LoggerPowerBand.getInstance().registrarInfo(
                    "Archivo de grabacion creado: " + ruta);
        } catch (FileAlreadyExistsException excepcion) {
            if (!Files.isRegularFile(ruta)) {
                throw ArchivoException.rutaNoRegular(ruta.toString());
            }
        }
    }
}

package powerband.grabaciones;

import java.io.IOException;
import java.nio.channels.FileChannel;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.util.List;
import java.util.Objects;
import java.util.Optional;

import powerband.archivos.BuscadorIndice;
import powerband.archivos.EncabezadoGrabacion;
import powerband.archivos.EntradaIndice;
import powerband.archivos.EntradaListaGrabacion;
import powerband.archivos.ListaGrabaciones;
import powerband.archivos.ValidacionesGenerales;
import powerband.excepciones.ArchivoException;
import powerband.excepciones.ArchivoIncompletoException;
import powerband.util.Constantes;

public final class CargadorGrabaciones {

    private final ArchivosGrabacion archivos;

    public CargadorGrabaciones(ArchivosGrabacion archivos) {
        this.archivos = Objects.requireNonNull(archivos, "archivos");
    }
    //Lista de entradas
    public List<EntradaListaGrabacion> listar() throws IOException {
        return new ListaGrabaciones(archivos.archivoLista()).listar();
    }
    //Metodo de acceso a abrir un string desde afuera
    public Optional<GrabacionAbierta> abrir(String nombre)
            throws IOException {
        ValidacionesGenerales.validarNombre(nombre);
        //Llama al buscador que ya tiene el algoritmo
        Optional<EntradaIndice> entrada = new BuscadorIndice(
                archivos.archivoIndice()).buscar(nombre);
        if (entrada.isEmpty()) {
            return Optional.empty();
        }

        return Optional.of(abrir(entrada.get()));
    }
    //Abrir interno, recibe la entrada y devuelve la grabacion
    private GrabacionAbierta abrir(EntradaIndice entrada)
            throws IOException {
        Path rutaMusica = archivos.archivoMusica();
        FileChannel canal;
        //Abre el canal
        try {
            canal = FileChannel.open(rutaMusica, StandardOpenOption.READ);
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(
                    rutaMusica.toString(),
                    "abrir una grabación", excepcion);
        }
        try {
            long posicionInicial = entrada.posicionInicial();
            long posicionFinal = entrada.posicionFinal();
            long tamanoMusica;
            //Validacion de intervalos
            try {
                tamanoMusica = canal.size();
            } catch (IOException excepcion) {
                throw ArchivoException.errorES(
                        rutaMusica.toString(),
                        "obtener el tamaño de la música", excepcion);
            }
            validarIntervalo(rutaMusica, posicionInicial,
                    posicionFinal, tamanoMusica);
            //Lectura del encabezado
            byte[] datosEncabezado = GrabacionAbierta.leerBytes(
                    canal, rutaMusica, posicionInicial,
                    Constantes.TAMANO_ENCABEZADO_GRABACION);
            EncabezadoGrabacion encabezado;
           
            try {
                encabezado =
                        EncabezadoGrabacion.desdeBytes(datosEncabezado);
            } catch (IllegalArgumentException excepcion) {
                throw ArchivoException.registroCorrupto(
                        rutaMusica.toString(), excepcion);
            }
            //Revisa si el encabezado es correcto
            if (!encabezado.nombre().equals(entrada.nombre())
                    || encabezado.posicionInicial() != posicionInicial
                    || encabezado.posicionFinal() != posicionFinal) {
                throw ArchivoException.registroCorrupto(
                        rutaMusica.toString(),
                        new IllegalArgumentException(
                                "El encabezado no coincide con el índice"));
            }
            //Retorna la grabacion si paso todas las verficiaciones
            long posicionPrimerComando = posicionInicial
                    + Constantes.TAMANO_ENCABEZADO_GRABACION;
            return new GrabacionAbierta(
                    canal, rutaMusica, encabezado,
                    posicionPrimerComando, posicionFinal);
        } catch (IOException | RuntimeException | Error fallo) {
            try {
                canal.close();
            } catch (IOException falloCierre) {
                fallo.addSuppressed(falloCierre);
            }
            throw fallo;
        }
    }
    //Valida que el intervalo sea valido
    //Valida que el archivo tenga un tamaño apropiado y las posiciones sean posibles.
    private void validarIntervalo(Path rutaMusica,
            long posicionInicial, long posicionFinal, long tamanoMusica)
            throws IOException {
        // Los intervalos tiene sentido
        if (posicionInicial < 0 || posicionFinal < posicionInicial) {
            throw ArchivoException.registroCorrupto(
                    rutaMusica.toString(),
                    new IllegalArgumentException(
                            "El intervalo del índice no es válido"));
        }
        //No se sale del archivo
        if (posicionFinal > tamanoMusica) {
            throw ArchivoException.registroFueraDeRango(
                    rutaMusica.toString(), posicionFinal);
        }
        //El tamaño de los intervalos es suficiente para al menos 
        //tener el header
        long tamanoIntervalo = posicionFinal - posicionInicial;
        if (tamanoIntervalo
                < Constantes.TAMANO_ENCABEZADO_GRABACION) {
            throw new ArchivoIncompletoException(
                    rutaMusica.toString(), tamanoIntervalo,
                    Constantes.TAMANO_ENCABEZADO_GRABACION);
        }
        //El tamaño sin el header se puede dividir en comandos.
        long tamanoComandos = tamanoIntervalo
                - Constantes.TAMANO_ENCABEZADO_GRABACION;
        if (tamanoComandos % Constantes.TAMANO_COMANDO_GRABADO != 0) {
            throw new ArchivoIncompletoException(
                    rutaMusica.toString(), tamanoComandos,
                    Constantes.TAMANO_COMANDO_GRABADO);
        }
    }
}

package powerband.grabaciones;

import java.io.IOException;
import java.nio.ByteBuffer;
import java.nio.channels.FileChannel;
import java.nio.file.AtomicMoveNotSupportedException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.nio.file.StandardOpenOption;
import java.time.Instant;
import java.util.Objects;
import java.util.Optional;
import java.util.function.LongSupplier;

import powerband.archivos.ArchivoRegistros;
import powerband.archivos.BuscadorIndice;
import powerband.archivos.ComandoGrabado;
import powerband.archivos.EncabezadoGrabacion;
import powerband.archivos.EntradaIndice;
import powerband.archivos.EntradaListaGrabacion;
import powerband.archivos.ListaGrabaciones;
import powerband.archivos.OrdenadorIndice;
import powerband.archivos.ReemplazadorIndice;
import powerband.archivos.ValidacionesGenerales;
import powerband.excepciones.ArchivoException;
import powerband.excepciones.ArchivoIncompletoException;
import powerband.excepciones.NombreDuplicadoException;
import powerband.logging.LoggerPowerBand;
import powerband.util.Constantes;

public final class FinalizadorGrabacion {

    @FunctionalInterface
    interface ReemplazoIndice {
        void reemplazar(Path indice, Path resultado,
                long cantidadEsperada) throws IOException;
    }

    private final Grabador grabador;
    private final ArchivosGrabacion archivos;
    private final LongSupplier relojFecha;
    private final ReemplazoIndice reemplazoIndice;

    public FinalizadorGrabacion(Grabador grabador,
            ArchivosGrabacion archivos) {
        this(grabador, archivos, () -> Instant.now().toEpochMilli());
    }

    public FinalizadorGrabacion(Grabador grabador,
            ArchivosGrabacion archivos, LongSupplier relojFecha) {
        this(grabador, archivos, relojFecha,
                (indice, resultado, cantidadEsperada) ->
                        new ReemplazadorIndice(indice, resultado)
                                .reemplazar(cantidadEsperada));
    }

   protected FinalizadorGrabacion(Grabador grabador,
            ArchivosGrabacion archivos, LongSupplier relojFecha,
            ReemplazoIndice reemplazoIndice) {
        this.grabador = Objects.requireNonNull(
                grabador, "grabador");
        this.archivos = Objects.requireNonNull(
                archivos, "archivos");
        this.relojFecha = Objects.requireNonNull(
                relojFecha, "relojFecha");
        this.reemplazoIndice = Objects.requireNonNull(
                reemplazoIndice, "reemplazoIndice");
    }

    public void finalizar(String nombre) throws IOException {
        Path temporalGrabacion = grabador.archivoSesionTemporal();
        LimpiezaGrabacion limpieza =
                new LimpiezaGrabacion(temporalGrabacion, archivos);

        try {
            grabador.detener();
            ValidacionesGenerales.validarNombre(nombre);
            validarTemporal(temporalGrabacion);

            Optional<EntradaIndice> existente =
                    new BuscadorIndice(
                            archivos.archivoIndice()).buscar(nombre);
            if (existente.isPresent()) {
                throw new NombreDuplicadoException(
                        existente.get().nombre(), nombre,
                        archivos.archivoIndice().toString());
            }

            long posicionInicial =
                    Files.size(archivos.archivoMusica());
            long cantidadComandos = grabador.cantidadComandos();
            long posicionFinal = posicionInicial
                    + Constantes.TAMANO_ENCABEZADO_GRABACION
                    + cantidadComandos
                            * Constantes.TAMANO_COMANDO_GRABADO;

            long fecha = relojFecha.getAsLong();
            EncabezadoGrabacion encabezado =
                    new EncabezadoGrabacion(
                            nombre, posicionInicial,
                            posicionFinal, fecha);

            Path musicaTemporal = crearCopia(
                    archivos.archivoMusica(), "musica-final-");
            Path listaTemporal = null;
            Path indiceTemporal = null;
            try {
                listaTemporal = crearCopia(
                        archivos.archivoLista(), "lista-final-");
                indiceTemporal = crearCopia(
                        archivos.archivoIndice(), "indice-final-");
                agregarGrabacion(
                        musicaTemporal, temporalGrabacion,
                        encabezado, posicionInicial);
                new ListaGrabaciones(listaTemporal).agregar(
                        new EntradaListaGrabacion(nombre, fecha));

                agregarIndice(indiceTemporal, nombre,
                        posicionInicial, posicionFinal);

                Path indiceOrdenado =
                        new OrdenadorIndice(indiceTemporal).ordenar();
                try {
                    reemplazarTodos(
                            musicaTemporal, listaTemporal, indiceOrdenado,
                            limpieza);
                } finally {
                    eliminarTemporalesConRegistro(indiceOrdenado);
                }
                eliminarTemporalesConRegistro(
                        musicaTemporal, listaTemporal, indiceTemporal);
            } catch (IOException | RuntimeException | Error fallo) {
                eliminarTemporales(fallo, musicaTemporal,
                        listaTemporal, indiceTemporal);
                throw fallo;
            }
        } catch (IOException | RuntimeException | Error fallo) {
            limpieza.limpiar(fallo);
            throw fallo;
        }
    }

    private void validarTemporal(Path temporal) throws IOException {
        long tamano = Files.size(temporal);
        long residuo = tamano % Constantes.TAMANO_COMANDO_GRABADO;
        if (residuo != 0) {
            throw new ArchivoIncompletoException(
                    temporal.toString(), tamano,
                    Constantes.TAMANO_COMANDO_GRABADO);
        }

        try (ArchivoRegistros<ComandoGrabado> archivo =
                crearArchivoComandos(temporal)) {
            long cantidad = archivo.cantidadRegistros();
            for (long indice = 0; indice < cantidad; indice++) {
                archivo.leer(indice
                        * Constantes.TAMANO_COMANDO_GRABADO);
            }
        }
    }

    private void agregarGrabacion(Path musicaTemporal,
            Path temporalGrabacion,
            EncabezadoGrabacion encabezado,
            long posicionInicial) throws IOException {
        try (FileChannel musica = FileChannel.open(
                    musicaTemporal,
                    StandardOpenOption.READ,
                    StandardOpenOption.WRITE);
                ArchivoRegistros<ComandoGrabado> comandos =
                    crearArchivoComandos(temporalGrabacion)) {
            long posicion = posicionInicial;
            ByteBuffer encabezadoBytes =
                    ByteBuffer.wrap(encabezado.serializar());
            escribirCompletamente(musica, encabezadoBytes, posicion);
            posicion += Constantes.TAMANO_ENCABEZADO_GRABACION;

            long cantidad = comandos.cantidadRegistros();
            for (long indice = 0; indice < cantidad; indice++) {
                ComandoGrabado comando = comandos.leer(
                        indice * Constantes.TAMANO_COMANDO_GRABADO);
                ByteBuffer datos =
                        ByteBuffer.wrap(comando.serializar());
                escribirCompletamente(musica, datos, posicion);
                posicion += Constantes.TAMANO_COMANDO_GRABADO;
            }
        }
    }

    private void agregarIndice(Path indiceTemporal, String nombre,
            long posicionInicial, long posicionFinal)
            throws IOException {
        try (ArchivoRegistros<EntradaIndice> indice =
                crearArchivoIndice(indiceTemporal)) {
            long cantidad = indice.cantidadRegistros();
            indice.escribir(
                    cantidad * Constantes.TAMANO_ENTRADA_INDICE,
                    new EntradaIndice(
                            nombre, posicionInicial, posicionFinal));
        }
    }

    private void reemplazarTodos(Path musicaTemporal,
            Path listaTemporal, Path indiceOrdenado,
            LimpiezaGrabacion limpieza)
            throws IOException {
        Path musicaBackup = null;
        Path listaBackup = null;
        Path indiceBackup = null;
        boolean musicaReemplazada = false;
        boolean listaReemplazada = false;
        boolean indiceReemplazoIntentado = false;
        boolean conservarMusicaBackup = false;
        boolean conservarListaBackup = false;
        boolean conservarIndiceBackup = false;
        Throwable falloPrincipal = null;

        try {
            musicaBackup = crearCopia(
                    archivos.archivoMusica(), "musica-backup-");
            listaBackup = crearCopia(
                    archivos.archivoLista(), "lista-backup-");
            indiceBackup = crearCopia(
                    archivos.archivoIndice(), "indice-backup-");

            reemplazoAtomico(musicaTemporal, archivos.archivoMusica());
            musicaReemplazada = true;
            reemplazoAtomico(listaTemporal, archivos.archivoLista());
            listaReemplazada = true;

            long cantidadIndice;
            try (ArchivoRegistros<EntradaIndice> original =
                    crearArchivoIndice(archivos.archivoIndice())) {
                cantidadIndice = original.cantidadRegistros();
            }

             indiceReemplazoIntentado = true;
            reemplazoIndice.reemplazar(
                    archivos.archivoIndice(), indiceOrdenado,
                    cantidadIndice + 1);
            limpieza.limpiarExitosamente();
        } catch (IOException | RuntimeException | Error fallo) {
            falloPrincipal = fallo;
            if (indiceReemplazoIntentado) {
                conservarIndiceBackup = !restaurar(
                        indiceBackup, archivos.archivoIndice(), fallo);
            }
            if (musicaReemplazada) {
                conservarMusicaBackup = !restaurar(
                        musicaBackup, archivos.archivoMusica(), fallo);
            }
            if (listaReemplazada) {
                conservarListaBackup = !restaurar(
                        listaBackup, archivos.archivoLista(), fallo);
            }
            throw fallo;
        } finally {
            if (falloPrincipal == null) {
                eliminarTemporalesConRegistro(
                        musicaBackup, listaBackup, indiceBackup);
            } else {
                eliminarTemporales(falloPrincipal,
                        conservarMusicaBackup ? null : musicaBackup,
                        conservarListaBackup ? null : listaBackup,
                        conservarIndiceBackup ? null : indiceBackup);
            }
        }
    }

    private Path crearCopia(Path origen, String prefijo)
            throws IOException {
        Path destino = Files.createTempFile(
                origen.getParent(), prefijo, ".tmp");
        try {
            Files.copy(origen, destino,
                    StandardCopyOption.REPLACE_EXISTING);
            return destino;
        } catch (IOException | RuntimeException | Error fallo) {
            eliminarTemporales(fallo, destino);
            throw fallo;
        }
    }

    private void reemplazoAtomico(Path origen, Path destino)
            throws IOException {
        try {
            Files.move(origen, destino,
                    StandardCopyOption.ATOMIC_MOVE,
                    StandardCopyOption.REPLACE_EXISTING);
        } catch (AtomicMoveNotSupportedException excepcion) {
            throw new ArchivoException(
                    powerband.excepciones.tipos.TipoErrorArchivo
                            .REEMPLAZO_NO_ATOMICO,
                    destino.toString(), excepcion);
        }
    }

    private boolean restaurar(Path respaldo, Path destino,
            Throwable fallo) {
        try {
            reemplazoAtomico(respaldo, destino);
            return true;
        } catch (IOException | RuntimeException excepcion) {
            fallo.addSuppressed(excepcion);
            return false;
        }
    }

    private void eliminarTemporalesConRegistro(Path... temporales) {
        for (Path temporal : temporales) {
            try {
                Files.deleteIfExists(temporal);
            } catch (IOException | RuntimeException excepcion) {
                ArchivoException fallo = ArchivoException.errorES(
                        temporal.toString(),
                        "eliminar un respaldo temporal",
                        excepcion);
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        fallo.getMessage(),
                        temporal.toString());
            }
        }
    }

    private void eliminarTemporales(Throwable fallo,
            Path... temporales) {
        for (Path temporal : temporales) {
            if (temporal == null) {
                continue;
            }
            try {
                Files.deleteIfExists(temporal);
            } catch (IOException | RuntimeException excepcion) {
                fallo.addSuppressed(excepcion);
            }
        }
    }

    private void escribirCompletamente(FileChannel canal,
            ByteBuffer buffer, long posicion) throws IOException {
        while (buffer.hasRemaining()) {
            canal.write(buffer, posicion + buffer.position());
        }
    }

    private static ArchivoRegistros<ComandoGrabado>
            crearArchivoComandos(Path ruta) {
        return new ArchivoRegistros<>(
                ruta,
                Constantes.TAMANO_COMANDO_GRABADO,
                ComandoGrabado::desdeBytes,
                ComandoGrabado::serializar);
    }

    private static ArchivoRegistros<EntradaIndice>
            crearArchivoIndice(Path ruta) {
        return new ArchivoRegistros<>(
                ruta,
                Constantes.TAMANO_ENTRADA_INDICE,
                EntradaIndice::desdeBytes,
                EntradaIndice::serializar);
    }
}

    private final EncabezadoGrabacion encabezado;

        this.rutaMusica = Objects.requireNonNull(rutaMusica, "rutaMusica");

        this.siguientePosicion = siguientePosicion;

    }

    public EncabezadoGrabacion encabezado() {
        return encabezado;
    }

    public Optional<ComandoGrabado> siguienteComando()
            throws IOException {
        if (siguientePosicion == posicionFinal) {
            return Optional.empty();
        }

        byte[] datos = leerBytes(
                canal, rutaMusica, siguientePosicion,
                Constantes.TAMANO_COMANDO_GRABADO);
        ComandoGrabado comando;
        try {
            comando = ComandoGrabado.desdeBytes(datos);
        } catch (IllegalArgumentException excepcion) {
            throw ArchivoException.registroCorrupto(
                    rutaMusica.toString(), excepcion);
        }

        siguientePosicion += Constantes.TAMANO_COMANDO_GRABADO;
        return Optional.of(comando);
    }

    @Override
    public void close() throws IOException {
        try {
            canal.close();
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(
                    rutaMusica.toString(),
                    "cerrar la lectura de la grabación", excepcion);
        }
    }

    public static byte[] leerBytes(FileChannel canal, Path ruta,
            long posicion, int cantidad) throws IOException {
        ByteBuffer buffer = ByteBuffer.allocate(cantidad);
        try {
            while (buffer.hasRemaining()) {
                int leidos = canal.read(
                        buffer, posicion + buffer.position());
                if (leidos < 0) {
                    throw ArchivoException.registroFueraDeRango(
                            ruta.toString(),
                            posicion + buffer.position());
                }
                if (leidos == 0) {
                    throw ArchivoException.errorES(
                            ruta.toString(),
                            "leer registros de una grabación", null);
                }
            }
            return buffer.array();
        } catch (ArchivoException excepcion) {
            throw excepcion;
        } catch (IOException excepcion) {
            throw ArchivoException.errorES(
                    ruta.toString(),
                    "leer registros de una grabación", excepcion);
        }
    }
}

package powerband.grabaciones;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Objects;
import java.util.function.LongSupplier;

import powerband.archivos.ArchivoRegistros;
import powerband.archivos.ComandoGrabado;
import powerband.comandos.Comando;
import powerband.comandos.ObservadorComandos;
import powerband.excepciones.ArchivoException;
import powerband.excepciones.ArgumentoInvalidoException;
import powerband.excepciones.EstadoInvalidoException;
import powerband.logging.LoggerPowerBand;
import powerband.util.Constantes;

public final class Grabador implements ObservadorComandos {

    private static final long NANOS_POR_MILISEGUNDO = 1_000_000L;

    private final Path directorioGrabaciones;
    private final LongSupplier relojMonotonico;

    private final Object bloqueo = new Object();
    private boolean grabando;
    private long tiempoBaseNanos;
    private Path archivoSesionTemporal;
    private ArchivoRegistros<ComandoGrabado> archivoTemporal;
    private Exception falloGrabacion;

    public Grabador(Path directorioGrabaciones) {
        this(directorioGrabaciones, System::nanoTime);
    }

    public Grabador(Path directorioGrabaciones,
            LongSupplier relojMonotonico) {
        this.directorioGrabaciones = Objects.requireNonNull(
                directorioGrabaciones, "directorioGrabaciones")
                .toAbsolutePath().normalize();
        this.relojMonotonico = Objects.requireNonNull(
                relojMonotonico, "relojMonotonico");
    }

    public void iniciar() throws IOException {
        synchronized (bloqueo) {
            if (grabando) {
                throw EstadoInvalidoException.grabacionActiva();
            }

            Path temporal = null;
            try {
                Files.createDirectories(directorioGrabaciones);
                temporal = Files.createTempFile(
                        directorioGrabaciones,
                        "grabacion-", ".tmp");
                ArchivoRegistros<ComandoGrabado> archivo =
                        crearArchivoTemporal(temporal);
                if (archivo.cantidadRegistros() != 0) {
                    archivo.close();
                    throw ArchivoException.errorES(
                            temporal.toString(),
                            "iniciar una sesión temporal", null);
                }

                tiempoBaseNanos = relojMonotonico.getAsLong();
                archivoTemporal = archivo;
                archivoSesionTemporal = temporal;
                falloGrabacion = null;
                grabando = true;
            } catch (IOException excepcion) {
                if (temporal != null) {
                    eliminarTemporal(temporal, excepcion);
                }
                ArchivoException fallo = ArchivoException.errorES(
                        directorioGrabaciones.toString(),
                        "iniciar la grabación", excepcion);
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        fallo.getMessage(),
                        directorioGrabaciones.toString());
                throw fallo;
            } catch (RuntimeException excepcion) {
                if (temporal != null) {
                    eliminarTemporal(temporal, excepcion);
                }
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        excepcion.getMessage(),
                        directorioGrabaciones.toString());
                throw excepcion;
            }
        }
    }

    public void detener() throws IOException {
        synchronized (bloqueo) {
            if (!grabando) {
                throw EstadoInvalidoException.grabacionInactiva();
            }

            grabando = false;
            IOException falloCierre = null;
            try {
                archivoTemporal.close();
            } catch (IOException excepcion) {
                falloCierre = ArchivoException.errorES(
                        archivoSesionTemporal.toString(),
                        "cerrar la sesión de grabación", excepcion);
            }
            if (falloGrabacion != null) {
                if (falloCierre != null) {
                    falloGrabacion.addSuppressed(falloCierre);
                }
                if (falloGrabacion instanceof IOException excepcion) {
                    throw excepcion;
                }
                if (falloGrabacion instanceof RuntimeException excepcion) {
                    throw excepcion;
                }
                throw new IllegalStateException(
                        "Error inesperado durante la grabación",
                        falloGrabacion);
            }
            if (falloCierre != null) {
                throw falloCierre;
            }
        }
    }

    public boolean estaGrabando() {
        synchronized (bloqueo) {
            return grabando;
        }
    }

    public Path archivoSesionTemporal() {
        synchronized (bloqueo) {
            if (archivoSesionTemporal == null) {
                throw EstadoInvalidoException.grabacionInactiva();
            }
            return archivoSesionTemporal;
        }
    }

    public long cantidadComandos() throws IOException {
        synchronized (bloqueo) {
            if (archivoSesionTemporal == null) {
                throw EstadoInvalidoException.grabacionInactiva();
            }
            try (ArchivoRegistros<ComandoGrabado> archivo =
                    crearArchivoTemporal(archivoSesionTemporal)) {
                return archivo.cantidadRegistros();
            }
        }
    }

    //En este caso evita errores con excepciones para facilitar el manejo
    @Override
    public boolean comandoEnviado(Comando comando, long momentoNanos) {
        synchronized (bloqueo) {
            if (!grabando) {
                return false;
            }
            if (comando == null) {
                ArgumentoInvalidoException fallo =
                        ArgumentoInvalidoException.comandoNulo();
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        fallo.getMessage(),
                        archivoSesionTemporal.toString());
                return false;
            }

            long diferenciaNanos = momentoNanos - tiempoBaseNanos;
            if (diferenciaNanos < 0) {
                ArgumentoInvalidoException fallo =
                        ArgumentoInvalidoException.momentoAnterior();
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        fallo.getMessage(),
                        archivoSesionTemporal.toString());
                return false;
            }

            try {
                long posicion = cantidadRegistrosSinValidar()
                        * Constantes.TAMANO_COMANDO_GRABADO;
                ComandoGrabado registro = new ComandoGrabado(
                        diferenciaNanos / NANOS_POR_MILISEGUNDO,
                        comando.destino().id(),
                        comando.tipoAccion(),
                        comando.parametros());
                archivoTemporal.escribir(posicion, registro);
                return true;
            } catch (IOException excepcion) {
                ArchivoException fallo = ArchivoException.errorES(
                        archivoSesionTemporal.toString(),
                        "guardar un comando grabado", excepcion);
                registrarFalloGrabacion(fallo);
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        fallo.getMessage(),
                        archivoSesionTemporal.toString());
                return false;
            } catch (RuntimeException excepcion) {
                registrarFalloGrabacion(excepcion);
                LoggerPowerBand.getInstance().registrarErrorArchivo(
                        excepcion.getMessage(),
                        archivoSesionTemporal.toString());
                return false;
            }
        }
    }

    private void registrarFalloGrabacion(Exception fallo) {
        if (falloGrabacion == null) {
            falloGrabacion = fallo;
        }
    }

    private long cantidadRegistrosSinValidar() throws IOException {
        return archivoTemporal.cantidadRegistros();
    }

    private ArchivoRegistros<ComandoGrabado> crearArchivoTemporal(
            Path temporal) {
        return new ArchivoRegistros<>(
                temporal,
                Constantes.TAMANO_COMANDO_GRABADO,
                ComandoGrabado::desdeBytes,
                ComandoGrabado::serializar);
    }

    private void eliminarTemporal(Path temporal, Throwable fallo) {
        try {
            Files.deleteIfExists(temporal);
        } catch (IOException excepcionLimpieza) {
            fallo.addSuppressed(excepcionLimpieza);
        }
    }
}

package powerband.grabaciones;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Objects;

import powerband.excepciones.ArchivoException;
import powerband.logging.LoggerPowerBand;

public final class LimpiezaGrabacion {

    private final Path archivoTemporal;
    private final ArchivosGrabacion archivos;

    public LimpiezaGrabacion(Path archivoTemporal,
            ArchivosGrabacion archivos) {
        this.archivos = Objects.requireNonNull(
                archivos, "archivos");
        Path rutaTemporal = Objects.requireNonNull(
                archivoTemporal, "archivoTemporal")
                .toAbsolutePath().normalize();
        if (rutaTemporal.equals(archivos.archivoMusica())
                || rutaTemporal.equals(archivos.archivoLista())
                || rutaTemporal.equals(archivos.archivoIndice())) {
            throw new IllegalArgumentException(
                    "La ruta no es un temporal de grabación válido");
        }
        this.archivoTemporal = rutaTemporal;
    }

    public void limpiar(Throwable fallo) {
        Objects.requireNonNull(fallo, "fallo");
        try {
            eliminarTemporal();
        } catch (IOException | RuntimeException excepcionLimpieza) {
            fallo.addSuppressed(excepcionLimpieza);
        }

        try {
            String mensaje = fallo.getMessage();
            if (mensaje == null || mensaje.isBlank()) {
                mensaje = fallo.getClass().getSimpleName();
            }
            LoggerPowerBand.getInstance().registrarErrorArchivo(
                    mensaje, archivos.directorio().toString());
        } catch (RuntimeException falloRegistro) {
            fallo.addSuppressed(falloRegistro);
        }
    }

    public void limpiarExitosamente() throws IOException {
        try {
            eliminarTemporal();
        } catch (IOException | RuntimeException excepcionLimpieza) {
            throw ArchivoException.errorES(
                    archivoTemporal.toString(),
                    "eliminar el temporal de grabación",
                    excepcionLimpieza);
        }
    }

    private void eliminarTemporal() throws IOException {
        if (Files.isDirectory(archivoTemporal)) {
            throw ArchivoException.rutaNoRegular(
                    archivoTemporal.toString());
        }
        Files.deleteIfExists(archivoTemporal);
    }
}

package powerband.grabaciones;

import java.io.IOException;
import java.time.Duration;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.function.Function;
import java.util.function.LongSupplier;

import powerband.archivos.ComandoGrabado;
import powerband.comandos.Comando;
import powerband.comandos.TipoAccion;
import powerband.util.Constantes;

public final class Reproductor {

    private final Function<ComandoGrabado, Comando> reconstruir;
    private final LongSupplier relojMonotonico;
    private final EsperaReproduccion espera;
    private final Object bloqueoControl = new Object();
    private Thread hiloReproduccion;
    private Thread hiloEnEspera;
    private volatile boolean detencionSolicitada;

    public Reproductor(
            Function<ComandoGrabado, Comando> reconstruir,
            LongSupplier relojMonotonico,
            EsperaReproduccion espera) {
        this.reconstruir = Objects.requireNonNull(
                reconstruir, "reconstruir");
        this.relojMonotonico = Objects.requireNonNull(
                relojMonotonico, "relojMonotonico");
        this.espera = Objects.requireNonNull(espera, "espera");
    }

    public void reproducir(GrabacionAbierta grabacion)
            throws IOException, InterruptedException {
        Objects.requireNonNull(grabacion, "grabacion");
        try (GrabacionAbierta lectura = grabacion) {
            iniciarControl();
            try {
                reproducirLectura(lectura);
            } finally {
                terminarControl();
            }
        }
    }
    //Detiene el hilos
    public boolean detener() {
        Thread hiloAInterrumpir;
        synchronized (bloqueoControl) {
            if (hiloReproduccion == null) {
                return false;
            }
            detencionSolicitada = true;
            hiloAInterrumpir = hiloEnEspera;
        }
        if (hiloAInterrumpir != null) {
            hiloAInterrumpir.interrupt();
        }
        return true;
    }
    //Lee una grabacion abierta, 
    private void reproducirLectura(GrabacionAbierta lectura)
            throws IOException, InterruptedException {
        long inicioNanos = relojMonotonico.getAsLong();
        //Crea una cola 
        Deque<ComandoGrabado> buffer = new ArrayDeque<>(
                Constantes.CAPACIDAD_BUFFER_REPRODUCCION);
        List<Comando> notasActivas = new ArrayList<>();
        boolean finLectura = false;
        Throwable falloPrincipal = null;

        try {
            while (!finLectura || !buffer.isEmpty()) {
                if (detencionSolicitada) {
                    break;
                }
                //Agrega comandos al buffer si puede
                while (!finLectura && !detencionSolicitada
                        && buffer.size()
                                < Constantes.CAPACIDAD_BUFFER_REPRODUCCION) {
                    Optional<ComandoGrabado> siguiente =
                            lectura.siguienteComando();
                    if (siguiente.isEmpty()) {
                        finLectura = true;
                    } else {
                        buffer.addLast(siguiente.orElseThrow());
                    }
                }
                //Toma elementos del buffer
                while (!buffer.isEmpty()) {
                    if (detencionSolicitada) {
                        break;
                    }
                    ComandoGrabado grabado = buffer.removeFirst();
                    //Espera el momento para ejecutar el comando
                    if (!esperarHasta(inicioNanos,
                            convertirTiempoANanos(
                                    grabado.tiempoRelativoMillis()))) {
                        break;
                    }
                    //Si se pide detener para aqui
                    if (detencionSolicitada) {
                        break;
                    }
                    //Reconstruye el comando
                    Comando comando = Objects.requireNonNull(
                            reconstruir.apply(grabado),
                            "La reconstrucción devolvió null");
                    validarComandoSostenido(comando);
                    if (detencionSolicitada) {
                        break;
                    }
                    //Ejecuta el comando en el sinteitzador
                    comando.ejecutar();
                    actualizarNotasActivas(comando, notasActivas);
                }
            }
        } catch (IOException | InterruptedException | RuntimeException
                | Error fallo) {
            falloPrincipal = fallo;
            throw fallo;
        } finally {
            RuntimeException falloFinalizacion =
                    finalizarNotasActivas(notasActivas);
            if (falloFinalizacion != null) {
                if (falloPrincipal == null) {
                    throw falloFinalizacion;
                }
                if (falloPrincipal != falloFinalizacion) {
                    falloPrincipal.addSuppressed(falloFinalizacion);
                }
            }
        }
    }

    private boolean esperarHasta(long inicioNanos, long plazoNanos)
            throws InterruptedException {
        while (true) {
            if (detencionSolicitada) {
                return false;
            }
            if (Thread.interrupted()) {
                throw new InterruptedException(
                        "La reproducción fue interrumpida");
            }
            long transcurridoNanos =
                    relojMonotonico.getAsLong() - inicioNanos;
            if (transcurridoNanos < 0) {
                throw new IllegalStateException(
                        "El reloj monotónico retrocedió durante la reproducción");
            }
            if (transcurridoNanos >= plazoNanos) {
                return true;
            }
            Thread hiloActual = Thread.currentThread();
            synchronized (bloqueoControl) {
                if (detencionSolicitada) {
                    return false;
                }
                hiloEnEspera = hiloActual;
            }
            try {
                //Hace que el programa espere, no utiliza sleep
                espera.esperar(plazoNanos - transcurridoNanos);
            } catch (InterruptedException excepcion) {
                if (detencionSolicitada) {
                    return false;
                }
                throw excepcion;
            } finally {
                synchronized (bloqueoControl) {
                    if (hiloEnEspera == hiloActual) {
                        hiloEnEspera = null;
                    }
                }
            }
        }
    }
    //Asigna el thread
    private void iniciarControl() {
        synchronized (bloqueoControl) {
            if (hiloReproduccion != null) {
                throw new IllegalStateException(
                        "Ya hay una reproducción en curso");
            }
            detencionSolicitada = false;
            hiloReproduccion = Thread.currentThread();
        }
    }
    //Termina los hilos
    private void terminarControl() {
        synchronized (bloqueoControl) {
            if (hiloReproduccion == Thread.currentThread()
                    && detencionSolicitada) {
                Thread.interrupted();
            }
            hiloEnEspera = null;
            hiloReproduccion = null;
            detencionSolicitada = false;
        }
    }
    
    private static void validarComandoSostenido(Comando comando) {
        if (comando.tipoAccion() == TipoAccion.INICIO
                && comando.parametros().length == 0) {
            throw new IllegalArgumentException(
                    "Un comando de inicio requiere identificar la nota");
        }
    }
    //Actualiza la lista de notas activas
    private static void actualizarNotasActivas(Comando comando,
            List<Comando> notasActivas) {
        double[] parametros = comando.parametros();
        //Si se esta iniciando se agrega el comando
        if (comando.tipoAccion() == TipoAccion.INICIO) {
            notasActivas.add(comando);
        } else if (comando.tipoAccion() == TipoAccion.FINALIZACION
                && parametros.length > 0) {
            long notaFinalizada = Double.doubleToLongBits(parametros[0]);
            for (int indice = notasActivas.size() - 1;
                    indice >= 0; indice--) {
                Comando inicio = notasActivas.get(indice);
                double[] parametrosInicio = inicio.parametros();
                //Si la nota finalizo se elimina
                if (inicio.destino().id() == comando.destino().id()
                        && Double.doubleToLongBits(parametrosInicio[0])
                                == notaFinalizada) {
                    notasActivas.remove(indice);
                    break;
                }
            }
        }
    }

    private static RuntimeException finalizarNotasActivas(
            List<Comando> notasActivas) {
        RuntimeException falloPrincipal = null;
        for (int indice = notasActivas.size() - 1;
                indice >= 0; indice--) {
            Comando inicio = notasActivas.get(indice);
            try {
                Comando finalizacion = new ComandoFinalizacion(
                        inicio.destino(), inicio.parametros()[0]);
                finalizacion.ejecutar();
            } catch (RuntimeException fallo) {
                if (falloPrincipal == null) {
                    falloPrincipal = fallo;
                } else if (falloPrincipal != fallo) {
                    falloPrincipal.addSuppressed(fallo);
                }
            }
        }
        return falloPrincipal;
    }

    private static long convertirTiempoANanos(long tiempoRelativoMillis) {
        try {
            return Duration.ofMillis(tiempoRelativoMillis).toNanos();
        } catch (ArithmeticException excepcion) {
            throw new IllegalArgumentException(
                    "El tiempo relativo no puede representarse en nanosegundos: "
                            + tiempoRelativoMillis,
                    excepcion);
        }
    }
}

package powerband.input;

import java.util.Objects;
import java.util.function.Consumer;

import powerband.logging.LoggerPowerBand;

public abstract class Entrada implements AutoCloseable {

    private final String nombre;
    private volatile boolean activa;
    private volatile Thread hilo;
    private volatile Consumer<Entrada> callbackDesconexion;
    private volatile Runnable callbackRunnable;
    private volatile ObservadorDesconexion observadorDesconexion;

    protected Entrada(String nombre) {
        this.nombre = validarNombre(nombre);
    }

    protected Entrada(String nombre, Consumer<Entrada> callbackDesconexion) {
        this.nombre = validarNombre(nombre);
        this.callbackDesconexion = callbackDesconexion;
    }

    protected Entrada(String nombre, Runnable callbackRunnable) {
        this.nombre = validarNombre(nombre);
        this.callbackRunnable = callbackRunnable;
    }

    public String nombre() {
        return nombre;
    }

    public synchronized void iniciar() {
        if (activa) {
            return;
        }
        activa = true;
        hilo = new Thread(this::ejecutarHilo,
                "entrada-" + nombre);
        hilo.setDaemon(true);
        hilo.start();
    }

    public synchronized void detener() {
        activa = false;
        Thread hiloActual = hilo;
        if (hiloActual != null) {
            hiloActual.interrupt();
        }
        liberarRecursos();
    }

    public boolean estaActiva() {
        return activa;
    }

    public boolean activa() {
        return estaActiva();
    }

    public boolean estaIniciada() {
        return estaActiva();
    }

    public void iniciarHilo() {
        iniciar();
    }

    public void detenerHilo() {
        detener();
    }

    public void cambiarCallbackDesconexion(Consumer<Entrada> callback) {
        callbackDesconexion = callback;
    }

    public void registrarCallbackDesconexion(Consumer<Entrada> callback) {
        cambiarCallbackDesconexion(callback);
    }

    public void registrarCallbackDesconexion(Runnable callback) {
        callbackRunnable = callback;
    }

    public void registrarObservadorDesconexion(
            ObservadorDesconexion observador) {
        observadorDesconexion = observador;
    }

    public void desconectar(String motivo) {
        detener();
        String mensaje = "Desconexion de entrada " + nombre
                + ": " + (motivo == null ? "dispositivo desconectado" : motivo);
        LoggerPowerBand.getInstance().registrarError(mensaje);
        Consumer<Entrada> callback = callbackDesconexion;
        if (callback != null) {
            try {
                callback.accept(this);
            } catch (RuntimeException excepcion) {
                LoggerPowerBand.getInstance().registrarError(
                        "Error al notificar desconexion de " + nombre
                                + ": " + excepcion);
            }
        }
        ObservadorDesconexion observador = observadorDesconexion;
        if (observador != null) {
            try {
                observador.entradaDesconectada(this, motivo);
            } catch (RuntimeException excepcion) {
                LoggerPowerBand.getInstance().registrarError(
                        "Error al notificar desconexion de " + nombre
                                + ": " + excepcion);
            }
        }
        Runnable callbackSimple = callbackRunnable;
        if (callbackSimple != null) {
            try {
                callbackSimple.run();
            } catch (RuntimeException excepcion) {
                LoggerPowerBand.getInstance().registrarError(
                        "Error al notificar desconexion de " + nombre
                                + ": " + excepcion);
            }
        }
    }

    public void notificarDesconexion(String motivo) {
        desconectar(motivo);
    }

    public void rechazarValor(String descripcion, double valor,
                              double minimo, double maximo) {
        String mensaje = "Valor rechazado en " + nombre + ": "
                + descripcion + "=" + valor
                + " (rango permitido " + minimo + " a " + maximo + ")";
        LoggerPowerBand.getInstance().registrarError(mensaje);
        throw new IllegalArgumentException(mensaje);
    }

    protected void validarValor(String descripcion, double valor,

    }

            }

            }
        }
    }

    public static boolean esDispositivoValido(String nombre) {
        if (nombre == null) {
            return false;
        }
        String nombreNormalizado = nombre.toLowerCase(Locale.ROOT);
        return !nombreNormalizado.contains("keyboard")
                && !nombreNormalizado.contains("teclado")
                && !nombreNormalizado.contains("mouse")
                && !nombreNormalizado.contains("raton")
                && !nombreNormalizado.contains("ratón");
    }

    public static boolean esDispositivoValido(Controller controlador) {
        if (controlador == null) {
            return false;
        }
        Controller.Type tipo = controlador.getType();
        if (tipo == Controller.Type.KEYBOARD
                || tipo == Controller.Type.MOUSE) {
            return false;
        }
        return esDispositivoValido(controlador.getName());
    }

    public void seleccionarDispositivo(String nombre) {
        Objects.requireNonNull(nombre, "nombreDispositivo");
        if (!esDispositivoValido(nombre)) {
            throw new IllegalArgumentException(
                    "Teclado y mouse no son dispositivos validos");
        }
        nombreDispositivo = nombre;
    }

        validarValor("eje", eje,
                Constantes.VALOR_MUSICAL_MINIMO,
                Constantes.VALOR_MUSICAL_MAXIMO);
        validarValor("valor", valor,
                Constantes.VALOR_MUSICAL_MINIMO,
                Constantes.VALOR_MUSICAL_MAXIMO);
        if (control != null) {
            control.convertirEvento(eje, valor);
        }
    }

    @Override
    protected void ejecutarCaptura() throws Exception {
        dispositivo = buscarDispositivo();
        if (dispositivo == null) {
            throw new IllegalStateException(
                    "No se encontro el dispositivo de control configurado");
        }

        while (estaActiva()) {
            if (!dispositivo.poll()) {
                desconectar("el control dejo de responder");
                return;
            }
            EventQueue cola = dispositivo.getEventQueue();
            Event evento = new Event();
            while (cola.getNextEvent(evento)) {
                Component componente = evento.getComponent();
                if (componente == null) {
                    continue;
                }
                float valor = evento.getValue();
                double valorNormalizado =
                        NormalizadorEntrada.normalizarCeroUno(
                                valor, -1.0, 1.0);
                procesarEvento(
                        componente.getIdentifier().hashCode(),
                        valorNormalizado);
            }
            Thread.sleep(10);
        }
    }

    @Override
    protected void liberarRecursosEspecificos() {
        dispositivo = null;
    }

    private Controller buscarDispositivo() {
        Controller[] controladores =
                ControllerEnvironment.getDefaultEnvironment()
                        .getControllers();
        for (Controller controlador : controladores) {
            if (esDispositivoValido(controlador)
                    && (nombreDispositivo == null
                    || nombreDispositivo.equals(controlador.getName()))) {
                return controlador;
            }
        }
        return null;
    }
}

public class EntradaMicrofono extends Entrada {

    private volatile AudioDispatcher dispatcher;
    private final Microfono microfono;
    private final BiConsumer<Double, Double> receptor;

    public EntradaMicrofono() {
        this((Microfono) null, null, null);
    }

    public EntradaMicrofono(Microfono microfono) {

    }

    public EntradaMicrofono(Microfono microfono,
                            Consumer<Entrada> callbackDesconexion) {

    }

    public EntradaMicrofono(Microfono microfono,
                            Runnable callbackDesconexion) {
        super("microfono", callbackDesconexion);
        this.microfono = microfono;
        this.receptor = null;
    }

    public EntradaMicrofono(Microfono microfono,
                            BiConsumer<Double, Double> receptor,
                            Consumer<Entrada> callbackDesconexion) {
        super("microfono", callbackDesconexion);
        this.microfono = microfono;
        this.receptor = receptor;
    }

    public void procesarFrecuencia(double frecuenciaHz,
                                   double amplitud) {
        validarValor("frecuencia", frecuenciaHz,
                Constantes.FRECUENCIA_MINIMA_HZ,
                Constantes.FRECUENCIA_MAXIMA_HZ);
        validarValor("amplitud", amplitud,
                Constantes.AMPLITUD_MINIMA,
                Constantes.AMPLITUD_MAXIMA);
        if (receptor != null) {
            receptor.accept(frecuenciaHz, amplitud);
        } else if (microfono != null) {
            microfono.convertirFrecuencia(frecuenciaHz, amplitud);
        }
    }

    @Override
    protected void ejecutarCaptura() throws Exception {
        try {
            dispatcher = AudioDispatcherFactory.fromDefaultMicrophone(
                    Constantes.TAMANO_BUFFER_AUDIO,
                    Constantes.SUPERPOSICION_AUDIO);
        } catch (LineUnavailableException excepcion) {
            throw new IllegalStateException(
                    "No hay un microfono disponible", excepcion);
        }

        PitchProcessor.PitchEstimationAlgorithm algoritmo =
                PitchProcessor.PitchEstimationAlgorithm.FFT_YIN;
        dispatcher.addAudioProcessor(new PitchProcessor(
                algoritmo,
                44100.0f,
                Constantes.TAMANO_BUFFER_AUDIO,
                (resultado, evento) -> {
                    float frecuencia = resultado.getPitch();
                    if (frecuencia <= 0.0f || !resultado.isPitched()) {
                        return;
                    }
                    if (frecuencia > 0.0f) {
                        procesarFrecuencia(frecuencia,
                                Constantes.AMPLITUD_MAXIMA);
                    }
                }));
        dispatcher.run();
    }

    @Override
    protected void liberarRecursosEspecificos() {
        AudioDispatcher actual = dispatcher;
        if (actual != null) {
            actual.stop();
            dispatcher = null;
        }
    }
}

        Rectangle area = new Rectangle();
        for (GraphicsDevice dispositivo
                : GraphicsEnvironment.getLocalGraphicsEnvironment()
                        .getScreenDevices()) {
            area = area.union(dispositivo.getDefaultConfiguration()
                    .getBounds());
        }
        return area;

    @Override
    protected void ejecutarCaptura() throws Exception {
        try {
            Class<?> claseManager = Class.forName("jpen.PenManager");
            Canvas lienzo = new Canvas();
            managerJPen = claseManager
                    .getConstructor(java.awt.Component.class)
                    .newInstance(lienzo);
            Object pen = claseManager.getField("pen").get(managerJPen);
            Class<?> claseListener = Class.forName("jpen.event.PenListener");

            listenerJPen = Proxy.newProxyInstance(
                    claseListener.getClassLoader(),
                    new Class<?>[] {claseListener},
                    new ListenerJPen());
            pen.getClass().getMethod("addListener", claseListener)
                    .invoke(pen, listenerJPen);
            while (estaActiva()) {

            }
        }
    }

        Object manager = managerJPen;
        Object listener = listenerJPen;
        if (manager != null && listener != null) {
            try {
                Object pen = manager.getClass().getField("pen").get(manager);
                Class<?> claseListener =
                        Class.forName("jpen.event.PenListener");
                pen.getClass().getMethod("removeListener", claseListener)
                        .invoke(pen, listener);
            } catch (ReflectiveOperationException excepcion) {
                // La liberacion de JPen depende de recursos que pueden
                // haber sido desconectados por el sistema operativo.
            }
        }
        managerJPen = null;
        listenerJPen = null;

                        .getMethod("getLevelValue", claseTipo)
                        .invoke(pen, tipoX)).floatValue();

                        .getMethod("getLevelValue", claseTipo)
                        .invoke(pen, tipoY)).floatValue();
                float presion = ((Number) pen.getClass()
                        .getMethod("getLevelValue", claseTipo)
                        .invoke(pen, tipoPresion)).floatValue();
                procesarPosicion(x, y, presion);
            }
            return null;
        }
    }

package powerband.input;

public final class NormalizadorEntrada {

    private NormalizadorEntrada() {
    }

    public static double normalizar(double valor,
                                    double origenMinimo,
                                    double origenMaximo,
                                    double destinoMinimo,
                                    double destinoMaximo) {
        validarLimites(origenMinimo, origenMaximo,
                destinoMinimo, destinoMaximo);
        if (!Double.isFinite(valor)
                || valor < origenMinimo
                || valor > origenMaximo) {
            throw new IllegalArgumentException(
                    "Valor fuera del rango de entrada: " + valor);
        }

        double proporcion = (valor - origenMinimo)
                / (origenMaximo - origenMinimo);
        return destinoMinimo
                + proporcion * (destinoMaximo - destinoMinimo);
    }

    public static double normalizarCeroUno(double valor,
                                           double origenMinimo,
                                           double origenMaximo) {
        return normalizar(valor, origenMinimo, origenMaximo, 0.0, 1.0);
    }

    public static boolean estaEnRango(double valor,
                                      double minimo,
                                      double maximo) {
        return Double.isFinite(valor) && valor >= minimo && valor <= maximo;
    }

    public double normalizarValor(double valor,
                                  double origenMinimo,
                                  double origenMaximo,
                                  double destinoMinimo,
                                  double destinoMaximo) {
        return normalizar(valor, origenMinimo, origenMaximo,
                destinoMinimo, destinoMaximo);
    }

    private static void validarLimites(double origenMinimo,
                                       double origenMaximo,
                                       double destinoMinimo,
                                       double destinoMaximo) {
        if (!Double.isFinite(origenMinimo)
                || !Double.isFinite(origenMaximo)
                || !Double.isFinite(destinoMinimo)
                || !Double.isFinite(destinoMaximo)
                || origenMinimo >= origenMaximo) {
            throw new IllegalArgumentException(
                    "Los rangos de normalizacion no son validos");
        }
    }
}

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

import powerband.comandos.BusComandos;
import powerband.comandos.Comando;
import powerband.comandos.ComandoReconstruido;
import powerband.comandos.TipoAccion;
import powerband.logging.LoggerPowerBand;
import powerband.sintetizadores.Sintetizador;

public abstract class Instrumento {

    private Sintetizador sintetizador;
    private final List<Sintetizador> sintetizadores = new ArrayList<>();
    private final BusComandos busComandos;

    protected Instrumento() {
        this((Sintetizador) null, new BusComandos());
    }

    protected Instrumento(Sintetizador sintetizador) {
        this(sintetizador, new BusComandos());
    }

    protected Instrumento(BusComandos busComandos) {
        this((Sintetizador) null, busComandos);
    }

    protected Instrumento(Sintetizador sintetizador, BusComandos busComandos) {
        this.sintetizador = sintetizador;
        this.busComandos = Objects.requireNonNull(busComandos, "busComandos");
        if (sintetizador != null) {
            sintetizadores.add(sintetizador);
        }
    }

    protected Instrumento(List<? extends Sintetizador> sintetizadores,
                         BusComandos busComandos) {
        this.busComandos = Objects.requireNonNull(busComandos, "busComandos");
        Objects.requireNonNull(sintetizadores, "sintetizadores");
        for (Sintetizador destino : sintetizadores) {
            agregarSintetizador(destino);
        }
        this.sintetizador = this.sintetizadores.isEmpty()
                ? null
                : this.sintetizadores.get(0);
    }

    public Sintetizador sintetizador() {
        return sintetizador;
    }

    public Sintetizador obtenerSintetizador() {
        return sintetizador;
    }

    public Sintetizador reemplazarSintetizador(Sintetizador nuevoSintetizador) {
        Sintetizador anterior = sintetizador;
        sintetizador = nuevoSintetizador;
        if (nuevoSintetizador != null) {
            int indice = sintetizadores.indexOf(anterior);
            if (indice >= 0) {
                sintetizadores.set(indice, nuevoSintetizador);
            } else if (!sintetizadores.contains(nuevoSintetizador)) {
                sintetizadores.add(nuevoSintetizador);
            }
        } else if (anterior != null) {
            sintetizadores.remove(anterior);
        }
        return anterior;
    }

    public void asignarSintetizador(Sintetizador nuevoSintetizador) {
        reemplazarSintetizador(nuevoSintetizador);
    }

    public void cambiarSintetizador(Sintetizador nuevoSintetizador) {
        reemplazarSintetizador(nuevoSintetizador);
    }

    public void agregarSintetizador(Sintetizador nuevoSintetizador) {
        Objects.requireNonNull(nuevoSintetizador, "nuevoSintetizador");
        boolean existe = sintetizadores.stream()
                .anyMatch(destino -> destino == nuevoSintetizador);
        if (!existe) {
            sintetizadores.add(nuevoSintetizador);
        }
        if (sintetizador == null) {
            sintetizador = nuevoSintetizador;
        }
    }

    public void retirarSintetizador(Sintetizador sintetizadorRetirar) {
        Objects.requireNonNull(sintetizadorRetirar, "sintetizadorRetirar");
        sintetizadores.removeIf(destino -> destino == sintetizadorRetirar);
        if (sintetizador == sintetizadorRetirar) {
            sintetizador = sintetizadores.isEmpty()
                    ? null
                    : sintetizadores.get(0);
        }
    }

    public void reemplazarSintetizadores(
            List<? extends Sintetizador> nuevosSintetizadores) {
        Objects.requireNonNull(nuevosSintetizadores, "nuevosSintetizadores");
        sintetizadores.clear();
        for (Sintetizador destino : nuevosSintetizadores) {
            if (destino != null
                    && sintetizadores.stream()
                            .noneMatch(actual -> actual == destino)) {
                sintetizadores.add(destino);
            }
        }
        sintetizador = sintetizadores.isEmpty()
                ? null
                : sintetizadores.get(0);
    }

    public List<Sintetizador> sintetizadores() {
        return List.copyOf(sintetizadores);
    }

    public BusComandos busComandos() {
        return busComandos;
    }

    public void enviarAccion(TipoAccion tipoAccion, double... parametros) {
        if (sintetizadores.isEmpty()) {
            throw new IllegalStateException(
                    "El instrumento no tiene un sintetizador asignado");
        }
        for (Sintetizador destino : List.copyOf(sintetizadores)) {
            Comando comando;
            comando = new ComandoReconstruido(
                    destino, tipoAccion, parametros);
            enviarComando(comando);
        }
    }

    public void ejecutarAccion(TipoAccion tipoAccion, double... parametros) {
        enviarAccion(tipoAccion, parametros);
    }

    public void enviarComando(Comando comando) {
        Objects.requireNonNull(comando, "comando");
        comando.ejecutar();
        busComandos.publicar(comando, System.nanoTime());
    }

    protected void publicarComando(Comando comando) {
        enviarComando(comando);
    }

    protected void registrarError(String mensaje, RuntimeException excepcion) {
        LoggerPowerBand.getInstance().registrarError(
                mensaje + ": " + excepcion);
    }
}

        }
        double notaMidi = frecuenciaAMidi(frecuenciaHz);
        Comando comando = new ComandoPuntual(destino,
                notaMidi, amplitud);
        enviarComando(comando);
        return comando;
    }

    public Comando procesarFrecuencia(double frecuenciaHz,
                                      double amplitud) {
        return convertirFrecuencia(frecuenciaHz, amplitud);
    }

    public double frecuenciaAMidi(double frecuenciaHz) {
        if (!Double.isFinite(frecuenciaHz) || frecuenciaHz <= 0.0) {
            throw new IllegalArgumentException(
                    "La frecuencia debe ser finita y mayor que cero");
        }
        return 69.0 + Constantes.SEMITONOS_POR_OCTAVA * (
                Math.log(frecuenciaHz
                        / Constantes.FRECUENCIA_AFINACION_HZ)
                        / Math.log(2.0));
    }
}

    }

    public void registrarError(Object objeto) {

    }

        registrar(categoria, objeto, null);
    }

    private void registrar(CategoriaLog categoria, Object objeto, String ubicacion) {
        Objects.requireNonNull(categoria, "categoria");
        Objects.requireNonNull(objeto, "objeto");

        String mensaje = objeto.toString();
        if (mensaje.isBlank()) {
            throw new IllegalArgumentException("El evento no puede estar vacio");
        }

        StringBuilder linea = new StringBuilder();
        linea.append(Instant.now())
                .append(" [")
                .append(categoria.texto())
                .append("] ")
                .append(mensaje.replace('\n', ' ').replace('\r', ' ').strip());

        if (ubicacion != null && !ubicacion.isBlank()) {
            linea.append(" ubicacion=").append(ubicacion);
        }
        linea.append(System.lineSeparator());

        try {
            Files.createDirectories(carpetaLogs);
            Path archivo = carpetaLogs.resolve("powerband.log");
            Files.writeString(archivo, linea.toString(), StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.APPEND);
        } catch (IOException excepcion) {
            throw new IllegalStateException(
                    "No se pudo escribir el log en " + carpetaLogs, excepcion);
        }
    }
}

package powerband.sintetizadores;

import java.io.IOException;
import java.util.Objects;

import powerband.comandos.TipoAccion;
import powerband.logging.LoggerPowerBand;
import powerband.util.Constantes;

abstract class AdaptadorBandClient implements Sintetizador {

    protected final BandClient bandClient;

    AdaptadorBandClient(BandClient bandClient) {
        this.bandClient = Objects.requireNonNull(bandClient, "bandClient");
    }

    protected final void validarAccion(
            TipoAccion recibida, TipoAccion esperada, String sintetizador) {
        if (recibida != esperada) {
            rechazar(sintetizador + " no admite la accion " + recibida);
        }
    }

    protected final void validarCantidad(
            double[] parametros, int minima, int maxima, String sintetizador) {
        if (parametros == null
                || parametros.length < minima
                || parametros.length > maxima) {
            rechazar("Cantidad de parametros invalida para " + sintetizador);
        }
    }

    protected final int notaMidi(double valor, String sintetizador) {
        return enteroEnRango(valor, Constantes.NOTA_MIDI_MINIMA,
                Constantes.NOTA_MIDI_MAXIMA, "nota MIDI", sintetizador);
    }

    protected final int duracionMs(double valor, String sintetizador) {
        return enteroEnRango(valor, Constantes.DURACION_MINIMA_MS,
                Constantes.DURACION_MAXIMA_MS, "duracion en milisegundos",
                sintetizador);
    }

    protected final int enteroEnRango(
            double valor, int minimo, int maximo, String nombre, String sintetizador) {
        if (!Double.isFinite(valor)
                || valor != Math.rint(valor)
                || valor < minimo
                || valor > maximo) {
            rechazar(nombre + " fuera de rango para " + sintetizador + ": " + valor);
        }
        return (int) valor;
    }

    protected final float razonParcial(double valor, int numeroParcial) {
        float resultado = (float) valor;
        if (!Double.isFinite(valor) || valor <= 0.0
                || !Float.isFinite(resultado) || resultado <= 0.0f) {
            rechazar("La razon del parcial " + numeroParcial
                    + " debe ser positiva y finita");
        }
        return resultado;
    }

    protected final float amplitudParcial(double valor, int numeroParcial) {
        float resultado = (float) valor;
        if (!Double.isFinite(valor)
                || valor < Constantes.AMPLITUD_MINIMA
                || valor > Constantes.AMPLITUD_MAXIMA
                || !Float.isFinite(resultado)) {
            rechazar("La amplitud del parcial " + numeroParcial
                    + " debe estar entre 0 y 1");
        }
        return resultado;
    }

    protected final void enviar(String descripcion, OperacionTransporte operacion) {
        try {
            operacion.ejecutar();
        } catch (IOException fallo) {
            IllegalStateException error = new IllegalStateException(
                    "Fallo local al " + descripcion, fallo);
            registrarError(error);
            throw error;
        }
    }

    protected final void rechazar(String mensaje) {
        IllegalArgumentException error = new IllegalArgumentException(mensaje);
        registrarError(error);
        throw error;
    }

    private static void registrarError(RuntimeException error) {
        try {
            LoggerPowerBand.getInstance().registrarError(error.getMessage());
        } catch (IllegalStateException falloLog) {
            error.addSuppressed(falloLog);
        }
    }

    @FunctionalInterface
    protected interface OperacionTransporte {
        void ejecutar() throws IOException;
    }
}

package powerband.sintetizadores;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;

import powerband.util.Configuraciones;
import powerband.util.Constantes;

/**
 * OSC client that plays the synths defined in myband.sc
 * (classicPiano, stringBass, organ70s) on a running scsynth.
 *
 * It talks directly to scsynth using /s_new and /n_set, so no OSC library is
 * needed. The default destination comes from Configuraciones.
 */
public class BandClient implements AutoCloseable {

    private static final int ADD_TO_HEAD = 0;
    private static final int DEFAULT_GROUP = 1;
    private static final int FIRST_NODE_ID = 10000;

    private final DatagramSocket socket;
    private final InetAddress address;
    private final int port;

    private int nextNodeId = FIRST_NODE_ID;
    private final Map<Integer, Integer> activeOrgan = new HashMap<>(); // midi note -> node id

    public BandClient() throws IOException {
        this(Configuraciones.getInstance());
    }

    public BandClient(Configuraciones configuraciones) throws IOException {
        this(Objects.requireNonNull(configuraciones, "configuraciones").getHost(),
                configuraciones.getPuerto());
    }

    public BandClient(String host, int port) throws IOException {
        Objects.requireNonNull(host, "host");
        if (host.isBlank()) {
            throw new IllegalArgumentException("El host no puede estar vacio");
        }
        if (port < 1 || port > 65535) {
            throw new IllegalArgumentException("El puerto debe estar entre 1 y 65535");
        }
        this.address = InetAddress.getByName(host);
        this.socket = new DatagramSocket();
        this.port = port;
    }

    // ------------------------------------------------------------ synth API

    /** Classic piano chord (major) built on the given fundamental MIDI note. */
    public void playPiano(int note, int durMs) throws IOException {
        playPiano(note, durMs, true);
    }

    /** Classic piano chord, major or minor, built on the given fundamental. */
    public void playPiano(int note, int durMs, boolean major) throws IOException {
        sNew("classicPiano",
                "note", note,
                "durMs", durMs,
                "third", major ? 4 : 3);
    }

    /** String bass note with a duration in milliseconds. */
    public void playBass(int note, int durMs) throws IOException {
        sNew("stringBass",
                "note", note,
                "durMs", durMs);
    }

    /** 70's organ with default partials (1, 2, 3) and amplitudes. */
    public void noteOn(int note) throws IOException {
        noteOn(note,
                new float[] {
                        (float) Constantes.RAZON_ORGANO_1,
                        (float) Constantes.RAZON_ORGANO_2,
                        (float) Constantes.RAZON_ORGANO_3
                },
                new float[] {
                        (float) Constantes.AMPLITUD_ORGANO_1,
                        (float) Constantes.AMPLITUD_ORGANO_2,
                        (float) Constantes.AMPLITUD_ORGANO_3
                });
    }

    /** 70's organ with 3 partial ratios and 3 partial amplitudes. Sounds until noteOff. */
    public synchronized void noteOn(int note, float[] ratios, float[] amps) throws IOException {
        if (ratios == null || amps == null
                || ratios.length != Constantes.CANTIDAD_PARCIALES_ORGANO
                || amps.length != Constantes.CANTIDAD_PARCIALES_ORGANO) {
            throw new IllegalArgumentException("organ needs exactly 3 ratios and 3 amps");
        }
        noteOff(note); // retrigger safely
        int id = sNew("organ70s",
                "freq", midiToFreq(note),
                "r1", ratios[0], "r2", ratios[1], "r3", ratios[2],
                "a1", amps[0], "a2", amps[1], "a3", amps[2]);
        activeOrgan.put(note, id);
    }

    public synchronized void noteOff(int note) throws IOException {
        Integer id = activeOrgan.get(note);
        if (id != null) {
            send("/n_set", id, "gate", 0);
            activeOrgan.remove(note, id);
        }
    }

    public synchronized void allNotesOff() throws IOException {
        IOException falloEnvio = null;
        for (Integer note : new ArrayList<>(activeOrgan.keySet())) {
            try {
                noteOff(note);
            } catch (IOException fallo) {
                if (falloEnvio == null) {
                    falloEnvio = fallo;
                } else {
                    falloEnvio.addSuppressed(fallo);
                }
            }
        }
        if (falloEnvio != null) {
            throw falloEnvio;
        }
    }

    // ------------------------------------------------------------- helpers

    private static float midiToFreq(int note) {
        return (float) (Constantes.FRECUENCIA_AFINACION_HZ
                * Math.pow(2.0,
                        (note - 69) / (double) Constantes.SEMITONOS_POR_OCTAVA));
    }

    /** Sends /s_new defName nodeId addAction target [name value ...]; returns the node id. */
    private synchronized int sNew(String defName, Object... controls) throws IOException {
        int id = nextNodeId++;
        Object[] args = new Object[4 + controls.length];
        args[0] = defName;
        args[1] = id;
        args[2] = ADD_TO_HEAD;
        args[3] = DEFAULT_GROUP;
        System.arraycopy(controls, 0, args, 4, controls.length);
        send("/s_new", args);
        return id;
    }

    /** Encodes and sends one OSC message. Supports Integer, Float and String arguments. */
    private void send(String path, Object... args) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        writeString(out, path);

        StringBuilder tags = new StringBuilder(",");
        for (Object a : args) {
            if (a instanceof Integer) tags.append('i');
            else if (a instanceof Float) tags.append('f');
            else if (a instanceof String) tags.append('s');
            else throw new IllegalArgumentException("unsupported OSC type: " + a.getClass());
        }
        writeString(out, tags.toString());

        for (Object a : args) {
            if (a instanceof Integer) writeInt(out, (Integer) a);
            else if (a instanceof Float) writeInt(out, Float.floatToIntBits((Float) a));
            else writeString(out, (String) a);
        }

        byte[] data = out.toByteArray();
        socket.send(new DatagramPacket(data, data.length, address, port));
    }

    private static void writeInt(ByteArrayOutputStream out, int v) {
        out.write(v >>> 24);
        out.write(v >>> 16);
        out.write(v >>> 8);
        out.write(v);
    }

    /** OSC string: bytes + null terminator, padded to a multiple of 4. */
    private static void writeString(ByteArrayOutputStream out, String s) {
        byte[] b = s.getBytes(StandardCharsets.US_ASCII);
        out.write(b, 0, b.length);
        int pad = 4 - (b.length % 4); // always adds at least the terminating null
        for (int i = 0; i < pad; i++) out.write(0);
    }

    @Override
    public synchronized void close() throws IOException {
        if (socket.isClosed()) {
            return;
        }
        IOException falloEnvio = null;
        try {
            allNotesOff();
        } catch (IOException fallo) {
            falloEnvio = fallo;
        }
        socket.close();
        if (falloEnvio != null) {
            throw falloEnvio;
        }
    }

    // ---------------------------------------------------------------- test

    /**
     * Test: start SuperCollider, run ~myband.load; (server booted, synths added),
     * then run this main. Optional args: host port.
     */
    public static void main(String[] args) throws Exception {
        Configuraciones configuraciones = Configuraciones.getInstance();
        String host = args.length > 0 ? args[0] : configuraciones.getHost();
        int port = args.length > 1
                ? Integer.parseInt(args[1])
                : configuraciones.getPuerto();

        try (BandClient band = new BandClient(host, port)) {
            System.out.println("Connecting to scsynth at " + host + ":" + port);

            System.out.println("-- piano: C major, A minor, F major");
            band.playPiano(48, 1500);       Thread.sleep(1700);
            band.playPiano(57, 1500, false); Thread.sleep(1700);
            band.playPiano(53, 2000);       Thread.sleep(2300);

            System.out.println("-- string bass: C G A F");
            for (int n : new int[] { 36, 43, 45, 41 }) {
                band.playBass(n, 450);
                Thread.sleep(500);
            }
            Thread.sleep(500);

            System.out.println("-- organ 70s: noteOn / noteOff");
            band.noteOn(60);                                                    Thread.sleep(1500);
            band.noteOn(64, new float[] { 1f, 2.01f, 4f }, new float[] { 1f, 0.6f, 0.3f }); Thread.sleep(1500);
            band.noteOn(67, new float[] { 1f, 3f, 5f }, new float[] { 1f, 0.4f, 0.2f });    Thread.sleep(2000);
            band.noteOff(60);
            band.noteOff(64);
            System.out.println("-- organ 70s: allNotesOff");
            band.allNotesOff();
            Thread.sleep(1000);

            System.out.println("Done.");
        }
    }
}

package powerband.sintetizadores;

import java.util.HashSet;
import java.util.Set;

import powerband.comandos.TipoAccion;
import powerband.util.Constantes;

public final class OrganoSintetizador extends AdaptadorBandClient {

    private final Set<Integer> notasActivas = new HashSet<>();

    public OrganoSintetizador(BandClient bandClient) {
        super(bandClient);
    }

    @Override
    public IdSintetizador id() {
        return IdSintetizador.ORGANO;
    }

    @Override
    public synchronized void ejecutar(
            TipoAccion tipoAccion, double... parametros) {
        if (tipoAccion == TipoAccion.INICIO) {
            iniciarNota(parametros);
        } else if (tipoAccion == TipoAccion.FINALIZACION) {
            finalizarNota(parametros);
        } else {
            rechazar("El organo solo admite inicio y finalizacion de notas");
        }
    }

    @Override
    public synchronized void liberarTodasLasNotas() {
        IllegalStateException falloTotal = null;
        for (int nota : Set.copyOf(notasActivas)) {
            try {
                enviar("finalizar la nota " + nota + " del organo",
                        () -> bandClient.noteOff(nota));
                notasActivas.remove(nota);
            } catch (IllegalStateException fallo) {
                if (falloTotal == null) {
                    falloTotal = fallo;
                } else {
                    falloTotal.addSuppressed(fallo);
                }
            }
        }
        if (falloTotal != null) {
            throw falloTotal;
        }
    }

    private void iniciarNota(double[] parametros) {
        boolean personalizada = parametros != null
                && parametros.length == 1 + 2 * Constantes.CANTIDAD_PARCIALES_ORGANO;
        if (!personalizada && (parametros == null || parametros.length != 1)) {
            rechazar("El inicio del organo requiere una nota y, opcionalmente, "
                    + "tres razones y tres amplitudes");
        }

        int nota = notaMidi(parametros[0], "organo");
        float[] razones = new float[Constantes.CANTIDAD_PARCIALES_ORGANO];
        float[] amplitudes = new float[Constantes.CANTIDAD_PARCIALES_ORGANO];
        if (personalizada) {
            for (int indice = 0; indice < Constantes.CANTIDAD_PARCIALES_ORGANO; indice++) {
                razones[indice] = razonParcial(parametros[indice + 1], indice + 1);
                amplitudes[indice] = amplitudParcial(
                        parametros[indice + 1 + Constantes.CANTIDAD_PARCIALES_ORGANO],
                        indice + 1);
            }
        } else {
            razones = new float[] {
                    (float) Constantes.RAZON_ORGANO_1,
                    (float) Constantes.RAZON_ORGANO_2,
                    (float) Constantes.RAZON_ORGANO_3
            };
            amplitudes = new float[] {
                    (float) Constantes.AMPLITUD_ORGANO_1,
                    (float) Constantes.AMPLITUD_ORGANO_2,
                    (float) Constantes.AMPLITUD_ORGANO_3
            };
        }

        float[] razonesEnvio = razones;
        float[] amplitudesEnvio = amplitudes;
        if (notasActivas.contains(nota)) {
            enviar("finalizar la nota previa " + nota + " del organo",
                    () -> bandClient.noteOff(nota));
            notasActivas.remove(nota);
        }
        enviar("iniciar la nota " + nota + " del organo",
                () -> bandClient.noteOn(nota, razonesEnvio, amplitudesEnvio));
        notasActivas.add(nota);
    }

    private void finalizarNota(double[] parametros) {
        validarCantidad(parametros, 1, 1, "organo");
        int nota = notaMidi(parametros[0], "organo");
        if (notasActivas.contains(nota)) {
            enviar("finalizar la nota " + nota + " del organo",
                    () -> bandClient.noteOff(nota));
            notasActivas.remove(nota);
        }
    }
}

    public IdSintetizador id() {
        return IdSintetizador.PIANO;
    }

    @Override
    public void ejecutar(TipoAccion tipoAccion, double... parametros) {
        validarAccion(tipoAccion, TipoAccion.PUNTUAL, "piano");
        validarCantidad(parametros, 2, 3, "piano");

        int nota = notaMidi(parametros[0], "piano");
        int duracion = duracionMs(parametros[1], "piano");
        int tercera = parametros.length == 3
                ? enteroEnRango(parametros[2],
                        Constantes.TERCERA_MENOR_SEMITONOS,
                        Constantes.TERCERA_MAYOR_SEMITONOS,
                        "tercera", "piano")
                : Constantes.TERCERA_MAYOR_SEMITONOS;
        if (tercera != Constantes.TERCERA_MENOR_SEMITONOS
                && tercera != Constantes.TERCERA_MAYOR_SEMITONOS) {
            rechazar("La tercera del piano debe ser menor (3) o mayor (4)");
        }

        boolean mayor = tercera == Constantes.TERCERA_MAYOR_SEMITONOS;
        enviar("enviar el acorde de piano",
                () -> bandClient.playPiano(nota, duracion, mayor));
    }

    }

        JPanel acciones = new JPanel(new FlowLayout(

        acciones.setBackground(ConstantesGUI.COLOR_BLANCO);
        acciones.setBorder(BorderFactory.createCompoundBorder(

        botonConectar.setForeground(ConstantesGUI.COLOR_BLANCO);
        botonConectar.setBackground(ConstantesGUI.COLOR_ACENTO);
        botonConectar.setFocusPainted(false);
        botonConectar.addActionListener(
                evento -> ejecutarOperacion(controlador::conectar));
        acciones.add(botonConectar);

        botonDesconectar.setHorizontalAlignment(SwingConstants.CENTER);
        botonDesconectar.setFocusPainted(false);
        botonDesconectar.addActionListener(
                evento -> ejecutarOperacion(() -> {
                    controlador.desconectar();
                    return Boolean.TRUE;
                }));
        acciones.add(botonDesconectar);

        add(acciones, BorderLayout.CENTER);
    }

    JButton botonConectar() {
        return botonConectar;
    }

    JButton botonDesconectar() {
        return botonDesconectar;
    }

    FachadaControlador fachada() {
        return controlador;
    }

    private void ejecutarOperacion(
            Callable<Boolean> operacion) {
        if (operacionEnCurso) {
            return;
        }

        operacionEnCurso = true;
        actualizarDisponibilidad(false);
        new SwingWorker<Boolean, Void>() {
            @Override
            protected Boolean doInBackground() throws Exception {
                return operacion.call();
            }

            @Override
            protected void done() {
                try {
                    get();
                } catch (InterruptedException excepcion) {
                    Thread.currentThread().interrupt();
                    manejadorErrores.accept(excepcion);
                } catch (ExecutionException excepcion) {
                    Throwable causa = excepcion.getCause();
                    manejadorErrores.accept(
                            causa == null ? excepcion : causa);
                } finally {
                    operacionEnCurso = false;
                    actualizarDisponibilidad(true);
                }
            }
        }.execute();
    }

    private void actualizarDisponibilidad(boolean habilitados) {
        botonConectar.setEnabled(habilitados);
        botonDesconectar.setEnabled(habilitados);
    }

    private static void registrarError(Throwable fallo) {
        LoggerPowerBand.getInstance().registrarError(
                "Fallo en una operación del panel de conexión: " + fallo);
    }
}

                botonGuardar.setForeground(ConstantesGUI.COLOR_BLANCO);

            if (editor instanceof JCheckBox casilla) {
                return casilla.isSelected();

            String texto = ((JTextField) editor).getText();
            if (valorInicial instanceof String) {
                return texto;
            }
            if (valorInicial instanceof Integer) {
                return Integer.valueOf(texto);
            }
            if (valorInicial instanceof Long) {
                return Long.valueOf(texto);
            }
            if (valorInicial instanceof Double) {
                return Double.valueOf(texto);
            }
            throw new IllegalStateException(
                    "La opción no tiene un editor compatible: " + clave);

            if (editor instanceof JCheckBox casilla) {
                casilla.setSelected((Boolean) valorConfirmado);
            } else if (editor instanceof JTextField campo) {
                campo.setText(String.valueOf(valorConfirmado));
            }
        }

            return valor instanceof String
                    || valor instanceof Boolean
                    || valor instanceof Integer
                    || valor instanceof Long
                    || valor instanceof Double;

            if (valor instanceof Boolean valorBooleano) {
                return new JCheckBox("Activada", valorBooleano);

                    || valor instanceof Long
                    || valor instanceof Double) {

                        ConstantesGUI.COLUMNAS_CAMPO_CONFIGURACION);

            String tipo = valor == null
                    ? "null"
                    : valor.getClass().getSimpleName();
            String valorVisible = String.valueOf(valor);
            return new JLabel(
                    "Valor actual: " + valorVisible
                            + " (tipo " + tipo + "; edición no disponible)");

        acciones.add(botonIniciar);

        botonDetener.setFocusPainted(false);
        botonDetener.addActionListener(
                evento -> solicitarDetencion());
        acciones.add(botonDetener);

        add(acciones, BorderLayout.CENTER);
    }

    }

        return botonIniciar;
    }

        return botonDetener;

    private void iniciarGrabacion() {
        ejecutarOperacion(() -> {
            controlador.iniciarGrabacion();
            return Boolean.TRUE;
        }, resultado -> { }, manejadorErrores);
    }

    private void solicitarDetencion() {
        if (operacionEnCurso) {
            return;
        }

        String nombre;
        try {
            nombre = solicitadorNombre.get();
        } catch (RuntimeException fallo) {
            manejadorErrores.accept(fallo);
            return;
        }
        if (nombre == null) {
            return;
        }

        ejecutarOperacion(() -> {
            List<String> coincidencias = Objects.requireNonNull(
                    controlador.buscarSesiones(nombre),
                    "buscarSesiones");
            if (!coincidencias.isEmpty()) {
                return Boolean.FALSE;
            }
            controlador.detenerGrabacion(nombre);
            return Boolean.TRUE;
        }, finalizada -> {
            if (!finalizada) {
                avisadorDuplicado.accept(nombre);
                solicitarDetencion();
            }
        }, fallo -> {
            if (fallo instanceof NombreDuplicadoException) {
                avisadorDuplicado.accept(nombre);
            } else {
                manejadorErrores.accept(fallo);
            }
        });
    }

    private <T> void ejecutarOperacion(
            Callable<T> operacion,
            Consumer<T> alCompletar,
            Consumer<Throwable> alFallar) {
        if (operacionEnCurso) {
            return;
        }

        operacionEnCurso = true;
        actualizarDisponibilidad(false);
        new SwingWorker<T, Void>() {
            @Override
            protected T doInBackground() throws Exception {
                return operacion.call();
            }

            @Override
            protected void done() {
                T resultado = null;
                Throwable fallo = null;
                try {
                    resultado = get();
                } catch (InterruptedException excepcion) {
                    Thread.currentThread().interrupt();
                    fallo = excepcion;
                } catch (ExecutionException excepcion) {
                    fallo = excepcion.getCause() == null
                            ? excepcion
                            : excepcion.getCause();
                } finally {
                    operacionEnCurso = false;
                    actualizarDisponibilidad(true);
                }

                if (fallo == null) {
                    alCompletar.accept(resultado);
                } else {
                    alFallar.accept(fallo);
                }
            }
        }.execute();
    }

    private void actualizarDisponibilidad(boolean habilitados) {
        botonIniciar.setEnabled(habilitados);
        botonDetener.setEnabled(habilitados);
    }

    private static String solicitarNombre() {
        return JOptionPane.showInputDialog(
                null,
                "Nombre de la grabación:",
                "Finalizar grabación",
                JOptionPane.QUESTION_MESSAGE);
    }

    private static void mostrarAvisoDuplicado(String nombre) {
        JOptionPane.showMessageDialog(
                null,
                "Ya existe una grabación con el nombre \"" + nombre
                        + "\". Elige otro nombre.",
                "Nombre duplicado",
                JOptionPane.WARNING_MESSAGE);
    }

    private static void registrarError(Throwable fallo) {
        LoggerPowerBand.getInstance().registrarError(
                "Fallo en una operación del panel de grabación: " + fallo);
    }

    private void cargarSeleccion(
            Map<String, String> dispositivos,
            Map<String, List<IdSintetizador>> sintetizadores) {
        String instrumento = listaInstrumentos.getSelectedValue();
        if (instrumento == null) {
            actualizarDisponibilidad(false);
            return;
        }

        String dispositivo = dispositivos.get(instrumento);
        if (dispositivo == null) {
            selectorDispositivo.setSelectedItem("Sin dispositivo");
        } else {
            selectorDispositivo.setSelectedItem(dispositivo);
        }

        List<IdSintetizador> asignados =
                sintetizadores.getOrDefault(instrumento, List.of());
        listaSintetizadores.clearSelection();
        for (int indice = 0;
                indice < modeloSintetizadores.size();
                indice++) {
            if (asignados.contains(
                    modeloSintetizadores.getElementAt(indice))) {
                listaSintetizadores.addSelectionInterval(indice, indice);
            }
        }
        mensajeEstado.setText(dispositivo == null
                ? "Instrumento: " + instrumento + " (sin dispositivo)"
                : "Instrumento: " + instrumento
                        + " (dispositivo activo)");
        actualizarDisponibilidad(!operacionEnCurso);
    }

    private void asignarDispositivo() {
        String instrumento = listaInstrumentos.getSelectedValue();
        Object seleccionado = selectorDispositivo.getSelectedItem();
        if (instrumento == null || seleccionado == null
                || "Sin dispositivo".equals(seleccionado)) {
            return;
        }

        String dispositivo = (String) seleccionado;
        ejecutarOperacion(
                () -> controlador.asignarDispositivo(
                        instrumento, dispositivo),
                asignado -> {
                    if (asignado) {
                        mensajeEstado.setText(
                                "Instrumento activo: " + instrumento);
                    } else {
                        mensajeEstado.setText(
                                "No se pudo asignar el dispositivo.");
                    }
                },
                fallo -> {
                    mensajeEstado.setText(
                            "No se pudo asignar el dispositivo.");
                    manejadorErrores.accept(fallo);
                });
    }

    private void asignarSintetizadores() {

        }

        new SwingWorker<T, Void>() {

    private void cargarSesiones(
            Callable<List<String>> consulta,
            String mensajeListaVacia) {
        mensajeEstado.setText("Cargando sesiones...");
        ejecutarOperacion(() -> {
            List<String> sesiones = Objects.requireNonNull(
                    consulta.call(), "resultado de sesiones");
            for (String sesion : sesiones) {
                Objects.requireNonNull(sesion, "nombre de sesión");
            }
            return sesiones;
        }, sesiones -> {
            modeloSesiones.clear();
            for (String sesion : sesiones) {
                modeloSesiones.addElement(sesion);
            }
            mensajeEstado.setText(
                    sesiones.isEmpty()
                            ? mensajeListaVacia
                            : "Sesiones disponibles: " + sesiones.size());
        }, fallo -> {
            mensajeEstado.setText("No se pudieron cargar las sesiones.");
            manejadorErrores.accept(fallo);
        });
    }

    private void reproducirSeleccion() {
        String nombre = listaSesiones.getSelectedValue();
        if (nombre == null || operacionEnCurso) {
            return;
        }

        ejecutarOperacion(
                () -> controlador.reproducirSesion(nombre),
                iniciada -> {
                    if (!iniciada) {
                        mensajeEstado.setText(
                                "No se pudo iniciar la reproducción.");
                    }
                },
                fallo -> {
                    mensajeEstado.setText(
                            "No se pudo solicitar la reproducción.");
                    manejadorErrores.accept(fallo);
                });
    }

    private <T> void ejecutarOperacion(
            Callable<T> operacion,
            Consumer<T> alCompletar,
            Consumer<Throwable> alFallar) {
        if (operacionEnCurso) {
            return;
        }

        operacionEnCurso = true;
        actualizarDisponibilidad();
        new SwingWorker<T, Void>() {
            @Override
            protected T doInBackground() throws Exception {
                return operacion.call();
            }

            @Override
            protected void done() {
                T resultado = null;
                Throwable fallo = null;
                try {
                    resultado = get();
                } catch (InterruptedException excepcion) {
                    Thread.currentThread().interrupt();
                    fallo = excepcion;
                } catch (ExecutionException excepcion) {
                    fallo = excepcion.getCause() == null
                            ? excepcion
                            : excepcion.getCause();
                } finally {
                    operacionEnCurso = false;
                    actualizarDisponibilidad();
                }

                if (fallo == null) {
                    alCompletar.accept(resultado);
                } else {
                    alFallar.accept(fallo);
                }
            }
        }.execute();
    }

    private void actualizarDisponibilidad() {
        boolean habilitados = !operacionEnCurso;
        campoBusqueda.setEnabled(habilitados);
        botonBuscar.setEnabled(habilitados);
        botonMostrarTodas.setEnabled(habilitados);
        botonActualizar.setEnabled(habilitados);
        listaSesiones.setEnabled(habilitados);
        botonReproducir.setEnabled(
                habilitados && listaSesiones.getSelectedIndex() >= 0);
    }

    private static void registrarError(Throwable fallo) {
        LoggerPowerBand.getInstance().registrarError(
                "Fallo en una operación del panel de sesiones: " + fallo);
    }
}

import java.awt.Dimension;

import javax.swing.JFrame;

import powerband.logging.LoggerPowerBand;

public final class VentanaPrincipal extends JFrame {

    private final FachadaControlador controlador;
    private final Runnable manejadorCierreCompletado;
    private Runnable accionCierre = this::iniciarCierre;
    private boolean cierreEnCurso;
    private boolean cierreCompletado;

    public VentanaPrincipal(FachadaControlador controlador) {
        this(controlador, () -> { });
    }

    protected VentanaPrincipal(
            FachadaControlador controlador,
            Runnable manejadorCierreCompletado) {
        super("PowerBand");
        this.controlador = Objects.requireNonNull(
                controlador, "controlador");
        this.manejadorCierreCompletado = Objects.requireNonNull(
                manejadorCierreCompletado, "manejadorCierreCompletado");

        setDefaultCloseOperation(WindowConstants.DO_NOTHING_ON_CLOSE);
        setMinimumSize(new Dimension(
                ConstantesGUI.ANCHO_MINIMO_VENTANA,
                ConstantesGUI.ALTO_MINIMO_VENTANA));
        setSize(ConstantesGUI.ANCHO_INICIAL_VENTANA,
                ConstantesGUI.ALTO_INICIAL_VENTANA);
        setLocationByPlatform(true);
        setContentPane(new PanelPrincipal(controlador));
        addWindowListener(new WindowAdapter() {
            @Override
            public void windowClosing(WindowEvent evento) {
                accionCierre.run();
            }
        });
    }

    public FachadaControlador controlador() {
        return controlador;
    }

    public void establecerAccionCierre(Runnable accionCierre) {
        this.accionCierre = Objects.requireNonNull(
                accionCierre, "accionCierre");
    }

    public final void cerrar() {
        iniciarCierre();
    }

    public final void limpiarObservadorEstado() {
        PanelPrincipal panel =
                (PanelPrincipal) getContentPane();
        panel.limpiarEstado();
    }

    protected boolean cierreCompletado() {
        return cierreCompletado;
    }

    private void iniciarCierre() {
        if (cierreEnCurso) {
            return;
        }
        cierreEnCurso = true;
        limpiarObservadorEstado();
        new SwingWorker<Void, Void>() {
            @Override
            protected Void doInBackground() throws Exception {
                controlador.desconectar();
                return null;
            }

            @Override
            protected void done() {
                try {
                    get();
                } catch (InterruptedException excepcion) {
                    Thread.currentThread().interrupt();
                    registrarFallo(excepcion);
                } catch (ExecutionException excepcion) {
                    Throwable causa = excepcion.getCause();
                    registrarFallo(causa == null ? excepcion : causa);
                } finally {
                    boolean anterior = cierreCompletado;
                    cierreCompletado = true;
                    firePropertyChange(
                            "cierreCompletado",
                            anterior,
                            cierreCompletado);
                    manejadorCierreCompletado.run();
                    dispose();
                }
            }
        }.execute();
    }

    private static void registrarFallo(Throwable fallo) {
        LoggerPowerBand.getInstance().registrarError(
                "Fallo al cerrar PowerBand mediante el controlador: "
                        + fallo);
    }
}

package powerband.util;

import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;

public final class Configuraciones {

    private static final Configuraciones INSTANCIA = new Configuraciones();

    private String host;
    private int puerto;
    private final Map<String, Object> opcionesModificables;

    public Configuraciones() {
        this.host = Constantes.HOST_DEFECTO;
        this.puerto = Constantes.PUERTO_DEFECTO;
        this.opcionesModificables = new LinkedHashMap<>();
    }

    public static Configuraciones getInstance() {
        return INSTANCIA;
    }

    public static Configuraciones obtenerInstancia() {
        return getInstance();
    }

    public String getHost() {
        return host;
    }

    public void setHost(String host) {
        Objects.requireNonNull(host, "host");
        if (host.isBlank()) {
            throw new IllegalArgumentException("El host no puede estar vacio");
        }
        this.host = host;
    }

    public int getPuerto() {
        return puerto;
    }

    public void setPuerto(int puerto) {
        if (puerto < 1 || puerto > 65535) {
            throw new IllegalArgumentException("El puerto debe estar entre 1 y 65535");
        }
        this.puerto = puerto;
    }

    public Object getOpcion(String clave) {
        return opcionesModificables.get(requireClave(clave));
    }

    public void setOpcion(String clave, Object valor) {
        opcionesModificables.put(requireClave(clave), valor);
    }

    public void eliminarOpcion(String clave) {
        opcionesModificables.remove(requireClave(clave));
    }

    public Map<String, Object> getOpcionesModificables() {
        return Collections.unmodifiableMap(new LinkedHashMap<>(opcionesModificables));
    }

    private static String requireClave(String clave) {
        Objects.requireNonNull(clave, "clave");
        if (clave.isBlank()) {
            throw new IllegalArgumentException("La clave no puede estar vacia");
        }
        return clave;
    }
}
